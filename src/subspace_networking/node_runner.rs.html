145    pub(crate) fn new(
146        NodeRunnerConfig {
147            allow_non_global_addresses_in_dht,
148            is_listening,
149            command_receiver,

152            next_random_query_interval,

156            libp2p_metrics,
157            metrics,
158            protocol_version,
159            bootstrap_addresses,
160        }: NodeRunnerConfig,

162        // Setup the address removal events exchange between persistent params storage and Kademlia.
163        let (removed_addresses_tx, removed_addresses_rx) = mpsc::unbounded();
164        let mut address_removal_task_handler_id = None;
165        if let Some(handler_id) = known_peers_registry.on_unreachable_address({
166            Arc::new(move |event| {
167                if let Err(error) = removed_addresses_tx.unbounded_send(event.clone()) {
168                    debug!(?error, ?event, "Cannot send PeerAddressRemovedEvent")
169                };
170            })
171        }) {
172            address_removal_task_handler_id.replace(handler_id);
173        }
174
175        Self {
176            allow_non_global_addresses_in_dht,
177            is_listening,
178            command_receiver,
179            swarm,
180            shared_weak,
181            next_random_query_interval,
182            query_id_receivers: HashMap::default(),
183            // We'll make the first query right away and continue at the interval.
184            random_query_timeout: Box::pin(tokio::time::sleep(Duration::from_secs(0)).fuse()),
185            // We'll make the first dial right away and continue at the interval.
186            periodical_tasks_interval: Box::pin(tokio::time::sleep(Duration::from_secs(0)).fuse()),
187            // There's not much point logging immediately, we can wait until we've bootstrapped.
188            last_peer_stats_info_log: Instant::now(),
189            known_peers_registry,
190            connected_servers: HashSet::new(),
191            reserved_peers,
192            temporary_bans,
193            libp2p_metrics,
194            metrics,
195            peer_ip_addresses: HashMap::new(),
196            protocol_version,
197            bootstrap_addresses,
198            bootstrap_command_state: Arc::new(AsyncMutex::new(BootstrapCommandState::default())),
199            removed_addresses_rx,
200            _address_removal_task_handler_id: address_removal_task_handler_id,
201        }
202    }

204    /// Drives the main networking future forward.
205    pub async fn run(&mut self) {
206        if self.is_listening {
207            // Wait for listen addresses, otherwise we will get ephemeral addresses in external address candidates that
208            // we do not want
209            loop {
210                if self.swarm.listeners().next().is_some() {
211                    break;
212                }
213
214                if let Some(swarm_event) = self.swarm.next().await {
215                    self.register_event_metrics(&swarm_event);
216                    self.handle_swarm_event(swarm_event).await;
217                } else {
218                    break;
219                }
220            }
221        }
222
223        self.bootstrap().await;
224
225        loop {
226            futures::select! {
227                _ = &mut self.random_query_timeout => {
228                    self.handle_random_query_interval();
229                    // Increase interval 2x, but limit it to MAX_RANDOM_QUERY_INTERVAL.
230                    self.random_query_timeout =
231                        Box::pin(tokio::time::sleep(self.next_random_query_interval).fuse());
232                    self.next_random_query_interval =
233                        (self.next_random_query_interval * 2).min(MAX_RANDOM_QUERY_INTERVAL);
234                },
235                swarm_event = self.swarm.next() => {
236                    if let Some(swarm_event) = swarm_event {
237                        self.register_event_metrics(&swarm_event);
238                        self.handle_swarm_event(swarm_event).await;
239                    } else {
240                        break;
241                    }
242                },
243                command = self.command_receiver.next() => {
244                    if let Some(command) = command {
245                        self.handle_command(command);
246                    } else {
247                        break;
248                    }
249                },
250                _ = self.known_peers_registry.run().fuse() => {
251                    trace!("Network parameters registry runner exited.")
252                },
253                _ = &mut self.periodical_tasks_interval => {
254                    self.handle_periodical_tasks().await;
255                    self.periodical_tasks_interval =
256                        Box::pin(tokio::time::sleep(PERIODICAL_TASKS_INTERVAL).fuse());
257                },
258                event = self.removed_addresses_rx.select_next_some() => {
259                    self.handle_removed_address_event(event);
260                },
261            }
262
263            // Allow to exit from busy loop during graceful shutdown
264            yield_now().await;
265        }
266    }

268    /// Bootstraps Kademlia network
269    async fn bootstrap(&mut self) {
270        // Add bootstrap nodes first to make sure there is space for them in k-buckets
271        for (peer_id, address) in strip_peer_id(self.bootstrap_addresses.clone()) {
272            self.swarm
273                .behaviour_mut()
274                .kademlia
275                .add_address(&peer_id, address);
276        }
277
278        let known_peers = self.known_peers_registry.all_known_peers().await;
279
280        if !known_peers.is_empty() {
281            for (peer_id, addresses) in known_peers {
282                for address in addresses.clone() {
283                    let address = match address.with_p2p(peer_id) {
284                        Ok(address) => address,
285                        Err(address) => {
286                            warn!(%peer_id, %address, "Failed to add peer ID to known peer address");
287                            break;
288                        }
289                    };
290                    self.swarm
291                        .behaviour_mut()
292                        .kademlia
293                        .add_address(&peer_id, address);
294                }
295
296                if let Err(error) = self
297                    .swarm
298                    .dial(DialOpts::peer_id(peer_id).addresses(addresses).build())
299                {
300                    warn!(%peer_id, %error, "Failed to dial peer during bootstrapping");
301                }
302            }
303
304            // Do bootstrap asynchronously
305            self.handle_command(Command::Bootstrap {
306                result_sender: None,
307            });
308            return;
309        }
310
311        let bootstrap_command_state = self.bootstrap_command_state.clone();
312        let mut bootstrap_command_state = bootstrap_command_state.lock().await;
313        let bootstrap_command_receiver = match &mut *bootstrap_command_state {
314            BootstrapCommandState::NotStarted => {
315                debug!("Bootstrap started.");
316
317                let (bootstrap_command_sender, bootstrap_command_receiver) = mpsc::unbounded();
318
319                self.handle_command(Command::Bootstrap {
320                    result_sender: Some(bootstrap_command_sender),
321                });
322
323                *bootstrap_command_state =
324                    BootstrapCommandState::InProgress(bootstrap_command_receiver);
325                match &mut *bootstrap_command_state {
326                    BootstrapCommandState::InProgress(bootstrap_command_receiver) => {
327                        bootstrap_command_receiver
328                    }
329                    _ => {
330                        unreachable!("Was just set to that exact value");
331                    }
332                }
333            }
334            BootstrapCommandState::InProgress(bootstrap_command_receiver) => {
335                bootstrap_command_receiver
336            }
337            BootstrapCommandState::Finished => {
338                return;
339            }
340        };
341
342        let mut bootstrap_step = 0;
343        loop {
344            futures::select! {
345                swarm_event = self.swarm.next() => {
346                    if let Some(swarm_event) = swarm_event {
347                        self.register_event_metrics(&swarm_event);
348                        self.handle_swarm_event(swarm_event).await;
349                    } else {
350                        break;
351                    }
352                },
353                result = bootstrap_command_receiver.next() => {
354                    if result.is_some() {
355                        debug!(%bootstrap_step, "Kademlia bootstrapping...");
356                        bootstrap_step += 1;
357                    } else {
358                        break;
359                    }
360                }
361            }
362        }
363
364        debug!("Bootstrap finished.");
365        *bootstrap_command_state = BootstrapCommandState::Finished;
366    }

368    /// Handles periodical tasks.
369    async fn handle_periodical_tasks(&mut self) {
370        // Log current connections.
371        let network_info = self.swarm.network_info();
372        let connections = network_info.connection_counters();
373
374        debug!(?connections, "Current DSN connections and limits.");
375
376        // Renew known external addresses.
377        let mut external_addresses = self.swarm.external_addresses().cloned().collect::<Vec<_>>();
378
379        if let Some(shared) = self.shared_weak.upgrade() {
380            debug!(?external_addresses, "Renew DSN external addresses.");
381            let mut addresses = shared.external_addresses.lock();
382            addresses.clear();
383            addresses.append(&mut external_addresses);
384        }
385
386        self.log_kademlia_stats();
387    }

400    fn handle_removed_address_event(&mut self, event: PeerAddressRemovedEvent) {
401        trace!(?event, "Peer address removed event.");
402
403        let bootstrap_node_ids = strip_peer_id(self.bootstrap_addresses.clone())
404            .into_iter()
405            .map(|(peer_id, _)| peer_id)
406            .collect::<Vec<_>>();
407
408        if bootstrap_node_ids.contains(&event.peer_id) {
409            debug!(
410                ?event,
411                ?bootstrap_node_ids,
412                "Skipped removing bootstrap node from Kademlia buckets."
413            );
414
415            return;
416        }
417
418        // Remove both versions of the address
419        self.swarm.behaviour_mut().kademlia.remove_address(
420            &event.peer_id,
421            &append_p2p_suffix(event.peer_id, event.address.clone()),
422        );
423
424        self.swarm
425            .behaviour_mut()
426            .kademlia
427            .remove_address(&event.peer_id, &remove_p2p_suffix(event.address));
428    }

430    fn handle_remove_listeners(&mut self, removed_listeners: &[Multiaddr]) {
431        let shared = match self.shared_weak.upgrade() {
432            Some(shared) => shared,
433            None => {
434                return;
435            }
436        };
437
438        // Remove both versions of the address
439        let peer_id = shared.id;
440        shared.listeners.lock().retain(|old_listener| {
441            !removed_listeners.contains(&append_p2p_suffix(peer_id, old_listener.clone()))
442                && !removed_listeners.contains(&remove_p2p_suffix(old_listener.clone()))
443        });
444    }

446    async fn handle_swarm_event(&mut self, swarm_event: SwarmEvent<Event>) {
447        match swarm_event {
448            SwarmEvent::Behaviour(Event::Identify(event)) => {
449                self.handle_identify_event(event).await;
450            }
451            SwarmEvent::Behaviour(Event::Kademlia(event)) => {
452                self.handle_kademlia_event(event).await;
453            }
454            SwarmEvent::Behaviour(Event::RequestResponse(event)) => {
455                self.handle_request_response_event(event).await;
456            }
457            SwarmEvent::Behaviour(Event::Autonat(event)) => {
458                self.handle_autonat_event(event).await;
459            }
460            ref event @ SwarmEvent::NewListenAddr { ref address, .. } => {
461                trace!(?event, "New local listener  event.");
462
463                let shared = match self.shared_weak.upgrade() {
464                    Some(shared) => shared,
465                    None => {
466                        return;
467                    }
468                };
469                shared.listeners.lock().push(address.clone());
470                shared.handlers.new_listener.call_simple(address);
471            }
472            ref event @ SwarmEvent::ListenerClosed { ref addresses, .. } => {
473                trace!(?event, "Local listener closed event.");
474                self.handle_remove_listeners(addresses);
475            }
476            ref event @ SwarmEvent::ExpiredListenAddr { ref address, .. } => {
477                trace!(?event, "Local listener expired event.");
478                self.handle_remove_listeners(slice::from_ref(address));
479            }
480            SwarmEvent::ConnectionEstablished {
481                peer_id,
482                endpoint,
483                num_established,
484                ..
485            } => {
486                // Save known addresses that were successfully dialed.
487                if let ConnectedPoint::Dialer { address, .. } = &endpoint {
488                    // filter non-global addresses when non-globals addresses are disabled
489                    if self.allow_non_global_addresses_in_dht || is_global_address_or_dns(address) {
490                        self.known_peers_registry
491                            .add_known_peer(peer_id, vec![address.clone()])
492                            .await;
493                    }
494                };
495
496                let shared = match self.shared_weak.upgrade() {
497                    Some(shared) => shared,
498                    None => {
499                        return;
500                    }
501                };
502
503                let is_reserved_peer = self.reserved_peers.contains_key(&peer_id);
504                debug!(
505                    %peer_id,
506                    %is_reserved_peer,
507                    ?endpoint,
508                    %num_established,
509                    "Connection established"
510                );
511
512                let maybe_remote_ip =
513                    endpoint
514                        .get_remote_address()
515                        .iter()
516                        .find_map(|protocol| match protocol {
517                            Protocol::Ip4(ip) => Some(IpAddr::V4(ip)),
518                            Protocol::Ip6(ip) => Some(IpAddr::V6(ip)),
519                            _ => None,
520                        });
521                if let Some(ip) = maybe_remote_ip {
522                    self.peer_ip_addresses
523                        .entry(peer_id)
524                        .and_modify(|ips| {
525                            ips.insert(ip);
526                        })
527                        .or_insert(HashSet::from([ip]));
528                }
529
530                let num_established_peer_connections = shared
531                    .num_established_peer_connections
532                    .fetch_add(1, Ordering::SeqCst)
533                    + 1;
534
535                shared
536                    .handlers
537                    .num_established_peer_connections_change
538                    .call_simple(&num_established_peer_connections);
539
540                // A new connection
541                if num_established.get() == 1 {
542                    shared.handlers.connected_peer.call_simple(&peer_id);
543                }
544
545                if let Some(metrics) = self.metrics.as_ref() {
546                    metrics.inc_established_connections()
547                }
548            }
549            SwarmEvent::ConnectionClosed {
550                peer_id,
551                num_established,
552                cause,
553                ..
554            } => {
555                let shared = match self.shared_weak.upgrade() {
556                    Some(shared) => shared,
557                    None => {
558                        return;
559                    }
560                };
561
562                debug!(
563                    %peer_id,
564                    ?cause,
565                    %num_established,
566                    "Connection closed with peer"
567                );
568
569                if num_established == 0 {
570                    self.peer_ip_addresses.remove(&peer_id);
571                    self.connected_servers.remove(&peer_id);
572                }
573                let num_established_peer_connections = shared
574                    .num_established_peer_connections
575                    .fetch_sub(1, Ordering::SeqCst)
576                    - 1;
577
578                shared
579                    .handlers
580                    .num_established_peer_connections_change
581                    .call_simple(&num_established_peer_connections);
582
583                // No more connections
584                if num_established == 0 {
585                    shared.handlers.disconnected_peer.call_simple(&peer_id);
586                }
587
588                if let Some(metrics) = self.metrics.as_ref() {
589                    metrics.dec_established_connections()
590                };
591            }
592            SwarmEvent::OutgoingConnectionError { peer_id, error, .. } => {
593                if let Some(peer_id) = &peer_id {
594                    let should_ban_temporarily =
595                        self.should_temporary_ban_on_dial_error(peer_id, &error);
596
597                    trace!(%should_ban_temporarily, "Temporary bans conditions.");
598
599                    if should_ban_temporarily {
600                        self.temporary_bans.lock().create_or_extend(peer_id);
601                        debug!(%peer_id, ?error, "Peer was temporarily banned.");
602                    }
603                }
604
605                debug!(
606                    ?peer_id,
607                    ?error,
608                    "SwarmEvent::OutgoingConnectionError for peer."
609                );
610
611                match error {
612                    DialError::Transport(ref addresses) => {
613                        for (addr, _) in addresses {
614                            trace!(?error, ?peer_id, %addr, "SwarmEvent::OutgoingConnectionError (DialError::Transport) for peer.");
615                            if let Some(peer_id) = peer_id {
616                                self.known_peers_registry
617                                    .remove_known_peer_addresses(peer_id, vec![addr.clone()])
618                                    .await;
619                            }
620                        }
621                    }
622                    DialError::WrongPeerId { obtained, .. } => {
623                        trace!(?error, ?peer_id, obtained_peer_id=?obtained, "SwarmEvent::WrongPeerId (DialError::WrongPeerId) for peer.");
624
625                        if let Some(ref peer_id) = peer_id {
626                            let kademlia = &mut self.swarm.behaviour_mut().kademlia;
627                            let _ = kademlia.remove_peer(peer_id);
628                        }
629                    }
630                    _ => {
631                        trace!(?error, ?peer_id, "SwarmEvent::OutgoingConnectionError");
632                    }
633                }
634            }
635            SwarmEvent::NewExternalAddrCandidate { address } => {
636                trace!(%address, "External address candidate");
637            }
638            SwarmEvent::ExternalAddrConfirmed { address } => {
639                debug!(%address, "Confirmed external address");
640
641                let connected_peers = self.swarm.connected_peers().copied().collect::<Vec<_>>();
642                self.swarm.behaviour_mut().identify.push(connected_peers);
643            }
644            SwarmEvent::ExternalAddrExpired { address } => {
645                debug!(%address, "External address expired");
646
647                let connected_peers = self.swarm.connected_peers().copied().collect::<Vec<_>>();
648                self.swarm.behaviour_mut().identify.push(connected_peers);
649            }
650            other => {
651                trace!("Other swarm event: {:?}", other);
652            }
653        }
654    }

656    fn should_temporary_ban_on_dial_error(&self, peer_id: &PeerId, error: &DialError) -> bool {
657        // TODO: Replace with banning of addresses rather peer IDs if this helps
658        if true {
659            return false;
660        }
661
662        // Ban temporarily only peers without active connections.
663        if self.swarm.is_connected(peer_id) {
664            return false;
665        }
666
667        match &error {
668            DialError::Transport(addresses) => {
669                for (_, error) in addresses {
670                    match error {
671                        TransportError::MultiaddrNotSupported(_) => {
672                            return true;
673                        }
674                        TransportError::Other(_) => {
675                            // Ignore "temporary ban" errors
676                            if self.temporary_bans.lock().is_banned(peer_id) {
677                                return false;
678                            }
679                        }
680                    }
681                }
682                // Other errors that are not related to temporary bans
683                true
684            }
685            DialError::LocalPeerId { .. } => {
686                // We don't ban ourselves
687                debug!("Local peer dial attempt detected.");
688
689                false
690            }
691            DialError::NoAddresses => {
692                // Let's wait until we get addresses
693                true
694            }
695            DialError::DialPeerConditionFalse(_) => {
696                // These are local conditions, we don't need to ban remote peers
697                false
698            }
699            DialError::Aborted => {
700                // Seems like a transient event
701                false
702            }
703            DialError::WrongPeerId { .. } => {
704                // It's likely that peer was restarted with different identity
705                false
706            }
707            DialError::Denied { .. } => {
708                // We exceeded the connection limits or we hit a black listed peer
709                false
710            }
711        }
712    }

714    async fn handle_identify_event(&mut self, event: IdentifyEvent) {
715        let local_peer_id = *self.swarm.local_peer_id();
716
717        if let IdentifyEvent::Received {
718            peer_id, mut info, ..
719        } = event
720        {
721            debug!(?peer_id, protocols = ?info.protocols, "IdentifyEvent::Received");
722
723            // Check for network partition
724            if info.protocol_version != self.protocol_version {
725                debug!(
726                    %local_peer_id,
727                    %peer_id,
728                    local_protocol_version = %self.protocol_version,
729                    peer_protocol_version = %info.protocol_version,
730                    "Peer has different protocol version, banning temporarily",
731                );
732
733                self.temporary_bans.lock().create_or_extend(&peer_id);
734                // Forget about this peer until they upgrade
735                let _ = self.swarm.disconnect_peer_id(peer_id);
736                self.swarm.behaviour_mut().kademlia.remove_peer(&peer_id);
737                self.known_peers_registry
738                    .remove_all_known_peer_addresses(peer_id);
739
740                return;
741            }
742
743            // Remove temporary ban if there was any
744            self.temporary_bans.lock().remove(&peer_id);
745
746            if info.listen_addrs.len() > MAX_LISTEN_ADDRESSES {
747                debug!(
748                    %local_peer_id,
749                    %peer_id,
750                    "Node has reported more than {} addresses; it is identified by {} and {}",
751                    MAX_LISTEN_ADDRESSES, info.protocol_version, info.agent_version
752                );
753                info.listen_addrs.truncate(MAX_LISTEN_ADDRESSES);
754            }
755
756            let kademlia = &mut self.swarm.behaviour_mut().kademlia;
757            let full_kademlia_support = kademlia
758                .protocol_names()
759                .iter()
760                .all(|local_protocol| info.protocols.contains(local_protocol));
761
762            if full_kademlia_support {
763                let received_addresses = info
764                    .listen_addrs
765                    .into_iter()
766                    .filter(|address| {
767                        if self.allow_non_global_addresses_in_dht
768                            || is_global_address_or_dns(address)
769                        {
770                            true
771                        } else {
772                            trace!(
773                                %local_peer_id,
774                                %peer_id,
775                                %address,
776                                "Ignoring self-reported non-global address",
777                            );
778
779                            false
780                        }
781                    })
782                    .collect::<Vec<_>>();
783                let received_address_strings = received_addresses
784                    .iter()
785                    .map(ToString::to_string)
786                    .collect::<Vec<_>>();
787                let old_addresses = kademlia
788                    .kbucket(peer_id)
789                    .and_then(|peers| {
790                        let key = peer_id.into();
791                        peers.iter().find_map(|peer| {
792                            (peer.node.key == &key).then_some(
793                                peer.node
794                                    .value
795                                    .iter()
796                                    .filter(|existing_address| {
797                                        let existing_address = existing_address.to_string();
798
799                                        !received_address_strings.iter().any(|received_address| {
800                                            received_address.starts_with(&existing_address)
801                                                || existing_address.starts_with(received_address)
802                                        })
803                                    })
804                                    .cloned()
805                                    .collect::<Vec<_>>(),
806                            )
807                        })
808                    })
809                    .unwrap_or_default();
810
811                for address in received_addresses {
812                    debug!(
813                        %local_peer_id,
814                        %peer_id,
815                        %address,
816                        protocol_names = ?kademlia.protocol_names(),
817                        "Adding self-reported address to Kademlia DHT",
818                    );
819
820                    kademlia.add_address(&peer_id, address);
821                }
822
823                for old_address in old_addresses {
824                    trace!(
825                        %local_peer_id,
826                        %peer_id,
827                        %old_address,
828                        "Removing old self-reported address from Kademlia DHT",
829                    );
830
831                    kademlia.remove_address(&peer_id, &old_address);
832                }
833
834                self.connected_servers.insert(peer_id);
835            } else {
836                debug!(
837                    %local_peer_id,
838                    %peer_id,
839                    peer_protocols = ?info.protocols,
840                    protocol_names = ?kademlia.protocol_names(),
841                    "Peer doesn't support our Kademlia DHT protocol",
842                );
843
844                kademlia.remove_peer(&peer_id);
845                self.connected_servers.remove(&peer_id);
846            }
847        }
848    }

850    async fn handle_kademlia_event(&mut self, event: KademliaEvent) {
851        trace!("Kademlia event: {:?}", event);
852
853        match event {
854            KademliaEvent::InboundRequest {
855                request: InboundRequest::AddProvider { record, .. },
856            } => {
857                debug!("Unexpected AddProvider request received: {:?}", record);
858            }
859            KademliaEvent::UnroutablePeer { peer } => {
860                debug!(%peer, "Unroutable peer detected");
861
862                self.swarm.behaviour_mut().kademlia.remove_peer(&peer);
863
864                if let Some(shared) = self.shared_weak.upgrade() {
865                    shared
866                        .handlers
867                        .peer_discovered
868                        .call_simple(&PeerDiscovered::UnroutablePeer { peer_id: peer });
869                }
870            }
871            KademliaEvent::RoutablePeer { peer, address } => {
872                debug!(?address, "Routable peer detected: {:?}", peer);
873
874                if let Some(shared) = self.shared_weak.upgrade() {
875                    shared
876                        .handlers
877                        .peer_discovered
878                        .call_simple(&PeerDiscovered::RoutablePeer {
879                            peer_id: peer,
880                            address,
881                        });
882                }
883            }
884            KademliaEvent::PendingRoutablePeer { peer, address } => {
885                debug!(?address, "Pending routable peer detected: {:?}", peer);
886
887                if let Some(shared) = self.shared_weak.upgrade() {
888                    shared
889                        .handlers
890                        .peer_discovered
891                        .call_simple(&PeerDiscovered::RoutablePeer {
892                            peer_id: peer,
893                            address,
894                        });
895                }
896            }
897            KademliaEvent::OutboundQueryProgressed {
898                step: ProgressStep { last, .. },
899                id,
900                result: QueryResult::GetClosestPeers(result),
901                ..
902            } => {
903                let mut cancelled = false;
904                if let Some(QueryResultSender::ClosestPeers { sender, .. }) =
905                    self.query_id_receivers.get(&id)
906                {
907                    match result {
908                        Ok(GetClosestPeersOk { key, peers }) => {
909                            trace!(
910                                "Get closest peers query for {} yielded {} results",
911                                hex::encode(key),
912                                peers.len(),
913                            );
914
915                            if peers.is_empty()
916                                // Connected peers collection is not empty.
917                                && self.swarm.connected_peers().next().is_some()
918                            {
919                                debug!("Random Kademlia query has yielded empty list of peers");
920                            }
921
922                            for peer in peers {
923                                cancelled = Self::unbounded_send_and_cancel_on_error(
924                                    &mut self.swarm.behaviour_mut().kademlia,
925                                    sender,
926                                    peer.peer_id,
927                                    "GetClosestPeersOk",
928                                    &id,
929                                ) || cancelled;
930                            }
931                        }
932                        Err(GetClosestPeersError::Timeout { key, peers }) => {
933                            debug!(
934                                "Get closest peers query for {} timed out with {} results",
935                                hex::encode(key),
936                                peers.len(),
937                            );
938
939                            for peer in peers {
940                                cancelled = Self::unbounded_send_and_cancel_on_error(
941                                    &mut self.swarm.behaviour_mut().kademlia,
942                                    sender,
943                                    peer.peer_id,
944                                    "GetClosestPeersError::Timeout",
945                                    &id,
946                                ) || cancelled;
947                            }
948                        }
949                    }
950                }
951
952                if last || cancelled {
953                    // There will be no more progress
954                    self.query_id_receivers.remove(&id);
955                }
956            }
957            KademliaEvent::OutboundQueryProgressed {
958                step: ProgressStep { last, count },
959                id,
960                result: QueryResult::Bootstrap(result),
961                stats,
962            } => {
963                debug!(?stats, %last, %count, ?id, ?result, "Bootstrap OutboundQueryProgressed step.");
964
965                let mut cancelled = false;
966                if let Some(QueryResultSender::Bootstrap { sender }) =
967                    self.query_id_receivers.get_mut(&id)
968                {
969                    match result {
970                        Ok(BootstrapOk {
971                            peer,
972                            num_remaining,
973                        }) => {
974                            trace!(%peer, %num_remaining, %last, "Bootstrap query step succeeded");
975
976                            cancelled = Self::unbounded_send_and_cancel_on_error(
977                                &mut self.swarm.behaviour_mut().kademlia,
978                                sender,
979                                (),
980                                "Bootstrap",
981                                &id,
982                            ) || cancelled;
983                        }
984                        Err(error) => {
985                            debug!(?error, "Bootstrap query failed.");
986                        }
987                    }
988                }
989
990                if last || cancelled {
991                    // There will be no more progress
992                    self.query_id_receivers.remove(&id);
993                }
994            }
995            _ => {}
996        }
997    }

999    // Returns `true` if query was cancelled
1000    fn unbounded_send_and_cancel_on_error<T>(
1001        kademlia: &mut Kademlia<DummyRecordStore>,
1002        sender: &mpsc::UnboundedSender<T>,
1003        value: T,
1004        channel: &'static str,
1005        id: &QueryId,
1006    ) -> bool {
1007        if sender.unbounded_send(value).is_err() {
1008            debug!("{} channel was dropped", channel);
1009
1010            // Cancel query
1011            if let Some(mut query) = kademlia.query_mut(id) {
1012                query.finish();
1013            }
1014            true
1015        } else {
1016            false
1017        }
1018    }

1025    async fn handle_autonat_event(&mut self, event: AutonatEvent) {
1026        trace!(?event, "Autonat event received.");
1027        let autonat = &self.swarm.behaviour().autonat;
1028        debug!(
1029            public_address=?autonat.public_address(),
1030            confidence=%autonat.confidence(),
1031            "Current public address confidence."
1032        );
1033
1034        match event {
1035            AutonatEvent::InboundProbe(_inbound_probe_event) => {
1036                // We do not care about this event
1037            }
1038            AutonatEvent::OutboundProbe(outbound_probe_event) => {
1039                match outbound_probe_event {
1040                    OutboundProbeEvent::Request { peer, .. } => {
1041                        // For outbound probe request add peer to allow list to ensure they can dial us back and not hit
1042                        // global incoming connection limit
1043                        self.swarm
1044                            .behaviour_mut()
1045                            .connection_limits
1046                            // We expect a single successful dial from this peer
1047                            .add_to_incoming_allow_list(
1048                                peer,
1049                                self.peer_ip_addresses
1050                                    .get(&peer)
1051                                    .iter()
1052                                    .flat_map(|ip_addresses| ip_addresses.iter())
1053                                    .copied(),
1054                                1,
1055                            );
1056                    }
1057                    OutboundProbeEvent::Response { peer, .. } => {
1058                        self.swarm
1059                            .behaviour_mut()
1060                            .connection_limits
1061                            .remove_from_incoming_allow_list(&peer, Some(1));
1062                    }
1063                    OutboundProbeEvent::Error { peer, .. } => {
1064                        if let Some(peer) = peer {
1065                            self.swarm
1066                                .behaviour_mut()
1067                                .connection_limits
1068                                .remove_from_incoming_allow_list(&peer, Some(1));
1069                        }
1070                    }
1071                }
1072            }
1073            AutonatEvent::StatusChanged { old, new } => {
1074                debug!(?old, ?new, "Public address status changed.");
1075
1076                // TODO: Remove block once https://github.com/libp2p/rust-libp2p/issues/4863 is resolved
1077                if let (NatStatus::Public(old_address), NatStatus::Private) = (old, new.clone()) {
1078                    self.swarm.remove_external_address(&old_address);
1079                    debug!(
1080                        ?old_address,
1081                        new_status = ?new,
1082                        "Removing old external address...",
1083                    );
1084
1085                    // Trigger potential mode change manually
1086                    self.swarm.behaviour_mut().kademlia.set_mode(None);
1087                }
1088
1089                let connected_peers = self.swarm.connected_peers().copied().collect::<Vec<_>>();
1090                self.swarm.behaviour_mut().identify.push(connected_peers);
1091            }
1092        }
1093    }

1095    fn handle_command(&mut self, command: Command) {
1096        match command {
1097            Command::GetClosestPeers {
1098                key,
1099                result_sender,
1100                permit,
1101            } => {
1102                let query_id = self.swarm.behaviour_mut().kademlia.get_closest_peers(key);
1103
1104                self.query_id_receivers.insert(
1105                    query_id,
1106                    QueryResultSender::ClosestPeers {
1107                        sender: result_sender,
1108                        _permit: permit,
1109                    },
1110                );
1111            }
1112            Command::GetClosestLocalPeers {
1113                key,
1114                source,
1115                result_sender,
1116            } => {
1117                let source = source.unwrap_or_else(|| *self.swarm.local_peer_id());
1118                let result = self
1119                    .swarm
1120                    .behaviour_mut()
1121                    .kademlia
1122                    .find_closest_local_peers(&KBucketKey::from(key), &source)
1123                    .filter(|peer| !peer.multiaddrs.is_empty())
1124                    .map(|peer| (peer.node_id, peer.multiaddrs))
1125                    .collect();
1126
1127                // Doesn't matter if receiver still waits for response.
1128                let _ = result_sender.send(result);
1129            }
1130            Command::GenericRequest {
1131                peer_id,
1132                addresses,
1133                protocol_name,
1134                request,
1135                result_sender,
1136            } => {
1137                self.swarm.behaviour_mut().request_response.send_request(
1138                    &peer_id,
1139                    protocol_name,
1140                    request,
1141                    result_sender,
1142                    IfDisconnected::TryConnect,
1143                    addresses,
1144                );
1145            }
1146            Command::BanPeer { peer_id } => {
1147                self.ban_peer(peer_id);
1148            }
1149            Command::Dial { address } => {
1150                let _ = self.swarm.dial(address);
1151            }
1152            Command::ConnectedPeers { result_sender } => {
1153                let connected_peers = self.swarm.connected_peers().cloned().collect();
1154
1155                let _ = result_sender.send(connected_peers);
1156            }
1157            Command::ConnectedServers { result_sender } => {
1158                let connected_servers = self.connected_servers.iter().cloned().collect();
1159
1160                let _ = result_sender.send(connected_servers);
1161            }
1162            Command::Bootstrap { result_sender } => {
1163                let kademlia = &mut self.swarm.behaviour_mut().kademlia;
1164
1165                match kademlia.bootstrap() {
1166                    Ok(query_id) => {
1167                        if let Some(result_sender) = result_sender {
1168                            self.query_id_receivers.insert(
1169                                query_id,
1170                                QueryResultSender::Bootstrap {
1171                                    sender: result_sender,
1172                                },
1173                            );
1174                        }
1175                    }
1176                    Err(err) => {
1177                        debug!(?err, "Bootstrap error.");
1178                    }
1179                }
1180            }
1181        }
1182    }

1184    fn ban_peer(&mut self, peer_id: PeerId) {
1185        // Remove temporary ban if there is one, before creating a permanent one.
1186        self.temporary_bans.lock().remove(&peer_id);
1187
1188        debug!(?peer_id, "Banning peer on network level");
1189
1190        self.swarm.behaviour_mut().block_list.block_peer(peer_id);
1191        self.swarm.behaviour_mut().kademlia.remove_peer(&peer_id);
1192        self.known_peers_registry
1193            .remove_all_known_peer_addresses(peer_id);
1194
1195        // Immediately disconnect the peer to cancel any in-flight requests.
1196        let _ = self.swarm.disconnect_peer_id(peer_id);
1197    }

1222    fn log_kademlia_stats(&mut self) {
1223        let mut kad_peers = 0;
1224        let mut kad_peers_with_no_address = 0;
1225
1226        for kbucket in self.swarm.behaviour_mut().kademlia.kbuckets() {
1227            for entry in kbucket.iter() {
1228                kad_peers += 1;
1229                if entry.node.value.len() == 0 {
1230                    kad_peers_with_no_address += 1;
1231                }
1232            }
1233        }
1234
1235        let (known_peers, known_peer_addresses) = self.known_peers_registry.count_known_peers();
1236
1237        let connected_peers = self.connected_servers.len();
1238
1239        let peers_with_ip_address = self.peer_ip_addresses.len();
1240        let peer_ip_address_count = self
1241            .peer_ip_addresses
1242            .values()
1243            .map(|addresses| addresses.len())
1244            .sum::<usize>();
1245
1246        debug!(
1247            %connected_peers,
1248            %kad_peers,
1249            %kad_peers_with_no_address,
1250            %known_peers,
1251            %known_peer_addresses,
1252            %peers_with_ip_address,
1253            %peer_ip_address_count,
1254            "dsn peers",
1255        );
1256
1257        if self.last_peer_stats_info_log.elapsed() >= PEER_INFO_LOG_INTERVAL {
1258            self.last_peer_stats_info_log = Instant::now();
1259            info!("dsn: actively using {connected_peers}/{kad_peers} known peers");
1260        }
