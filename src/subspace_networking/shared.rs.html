15/// Represents Kademlia events (RoutablePeer, PendingRoutablePeer, UnroutablePeer).
16#[derive(Clone, Debug)]
17pub enum PeerDiscovered {
18    /// Kademlia's unroutable peer event.
19    UnroutablePeer {
20        /// Peer ID
21        peer_id: PeerId,
22    },
23
24    /// Kademlia's routable or pending routable peer event.
25    RoutablePeer {
26        /// Peer ID
27        peer_id: PeerId,
28        /// Peer address
29        address: Multiaddr,
30    },
31}
32
33impl PeerDiscovered {
34    /// Extracts peer ID from event.
35    pub fn peer_id(&self) -> PeerId {
36        match self {
37            PeerDiscovered::UnroutablePeer { peer_id } => *peer_id,
38            PeerDiscovered::RoutablePeer { peer_id, .. } => *peer_id,
39        }
40    }
41}

43#[derive(Debug)]
44pub(crate) enum Command {
45    GetClosestPeers {
46        key: Multihash,
47        result_sender: mpsc::UnboundedSender<PeerId>,
48        permit: Option<OwnedSemaphorePermit>,
49    },
50    GetClosestLocalPeers {
51        key: Multihash,
52        source: Option<PeerId>,
53        result_sender: oneshot::Sender<Vec<(PeerId, Vec<Multiaddr>)>>,
54    },
55    GenericRequest {
56        peer_id: PeerId,
57        addresses: Vec<Multiaddr>,
58        protocol_name: &'static str,
59        request: Vec<u8>,
60        result_sender: oneshot::Sender<Result<Vec<u8>, RequestFailure>>,
61    },
62    BanPeer {
63        peer_id: PeerId,
64    },
65    Dial {
66        address: Multiaddr,
67    },
68    ConnectedPeers {
69        result_sender: oneshot::Sender<Vec<PeerId>>,
70    },
71    ConnectedServers {
72        result_sender: oneshot::Sender<Vec<PeerId>>,
73    },
74    Bootstrap {
75        // No result sender means background async bootstrapping
76        result_sender: Option<mpsc::UnboundedSender<()>>,
77    },
78}

80#[derive(Default, Debug)]
81pub(crate) struct Handlers {
82    pub(crate) new_listener: Handler<Multiaddr>,
83    pub(crate) num_established_peer_connections_change: Handler<usize>,
84    pub(crate) connected_peer: Handler<PeerId>,
85    pub(crate) disconnected_peer: Handler<PeerId>,
86    pub(crate) peer_discovered: Handler<PeerDiscovered>,
87}

89#[derive(Debug)]
90pub(crate) struct Shared {
91    pub(crate) handlers: Handlers,
92    pub(crate) id: PeerId,
93    /// Addresses on which node is listening for incoming requests.
94    pub(crate) listeners: Mutex<Vec<Multiaddr>>,
95    pub(crate) external_addresses: Mutex<Vec<Multiaddr>>,
96    pub(crate) num_established_peer_connections: Arc<AtomicUsize>,
97    /// Sender end of the channel for sending commands to the swarm.
98    pub(crate) command_sender: mpsc::Sender<Command>,
99    pub(crate) rate_limiter: RateLimiter,
100}
101
102impl Shared {
103    pub(crate) fn new(
104        id: PeerId,
105        command_sender: mpsc::Sender<Command>,
106        rate_limiter: RateLimiter,
107    ) -> Self {
108        Self {
109            handlers: Handlers::default(),
110            id,
111            listeners: Mutex::default(),
112            external_addresses: Mutex::default(),
113            num_established_peer_connections: Arc::new(AtomicUsize::new(0)),
114            command_sender,
115            rate_limiter,
116        }
117    }
