35/// Whether the transaction pool is notified about every imported block.
36///
37/// A single-state pool is only maintained on best blocks.
38fn all_block_notifications(options: &sc_transaction_pool::TransactionPoolOptions) -> bool {
39    options.use_all_block_notifications()
40}

42/// Spawn the tasks that are required to run a node.
43#[expect(clippy::result_large_err, reason = "Comes from Substrate")]
44pub(super) fn spawn_tasks<TBl, TBackend, TExPool, TRpc, TCl>(
45    params: SpawnTasksParams<TBl, TCl, TExPool, TRpc, TBackend>,
46) -> Result<RpcHandlers, Error>
47where
48    TCl: ProvideRuntimeApi<TBl>
49        + HeaderMetadata<TBl, Error = sp_blockchain::Error>
50        + Chain<TBl>
51        + BlockBackend<TBl>
52        + BlockIdTo<TBl, Error = sp_blockchain::Error>
53        + ProofProvider<TBl>
54        + HeaderBackend<TBl>
55        + BlockchainEvents<TBl>
56        + ExecutorProvider<TBl>
57        + UsageProvider<TBl>
58        + StorageProvider<TBl, TBackend>
59        + CallApiAt<TBl>
60        + Send
61        + 'static,
62    TCl::Api: sp_api::Metadata<TBl>
63        + sp_transaction_pool::runtime_api::TaggedTransactionQueue<TBl>
64        + sp_session::SessionKeys<TBl>
65        + sp_api::ApiExt<TBl>,
66    TBl: BlockT,
67    TBl::Hash: Unpin,
68    TBl::Header: Unpin,
69    TBackend: 'static + sc_client_api::backend::Backend<TBl> + Send,
70    TExPool: MaintainedTransactionPool<Block = TBl, Hash = TBl::Hash> + 'static,
71{
72    let SpawnTasksParams {
73        // TODO: Stop using `Configuration` once
74        //  https://github.com/paritytech/polkadot-sdk/pull/5364 is in our fork
75        mut config,
76        task_manager,
77        client,
78        backend,
79        keystore,
80        transaction_pool,
81        rpc_builder,
82        network,
83        system_rpc_tx,
84        tx_handler_controller,
85        sync_service,
86        telemetry,
87        tracing_execute_block,
88    } = params;
89
90    let chain_info = client.usage_info().chain;
91
92    let sysinfo = sc_sysinfo::gather_sysinfo();
93    sc_sysinfo::print_sysinfo(&sysinfo);
94
95    let telemetry = telemetry
96        .map(|telemetry| {
97            init_telemetry(
98                config.network.node_name.clone(),
99                config.impl_name.clone(),
100                config.impl_version.clone(),
101                config.chain_spec.name().to_string(),
102                config.role.is_authority(),
103                network.clone(),
104                client.clone(),
105                telemetry,
106                Some(sysinfo),
107            )
108        })
109        .transpose()?;
110
111    info!("📦 Highest known block at #{}", chain_info.best_number);
112
113    let spawn_handle = task_manager.spawn_handle();
114
115    // Inform the tx pool about imported and finalized blocks.
116    spawn_handle.spawn(
117        "txpool-notifications",
118        Some("transaction-pool"),
119        sc_transaction_pool::notification_future(
120            client.clone(),
121            transaction_pool.clone(),
122            all_block_notifications(&config.transaction_pool),
123        ),
124    );
125
126    spawn_handle.spawn(
127        "on-transaction-imported",
128        Some("transaction-pool"),
129        propagate_transaction_notifications(
130            transaction_pool.clone(),
131            tx_handler_controller,
132            telemetry.clone(),
133        ),
134    );
135
136    // Periodically updated metrics and telemetry updates.
137    spawn_handle.spawn(
138        "telemetry-periodic-send",
139        None,
140        MetricsService::new(telemetry).run(
141            client.clone(),
142            transaction_pool.clone(),
143            network.clone(),
144            sync_service.clone(),
145        ),
146    );
147
148    let rpc_id_provider = config.rpc.id_provider.take();
149
150    // A dedicated runtime for RPC tasks, matching upstream's spawn_tasks.
151    let rpc_runtime = sc_service::create_rpc_runtime(config.rpc.max_connections)?;
152    let rpc_spawn_handle: Arc<dyn sp_core::traits::SpawnNamed> = Arc::new(
153        sc_rpc_server::RpcSpawnHandle::new(rpc_runtime.handle().clone()),
154    );
155
156    // jsonrpsee RPC
157    let gen_rpc_module = || {
158        gen_rpc_module(sc_service::GenRpcModuleParams {
159            spawn_handle: rpc_spawn_handle.clone(),
160            client: client.clone(),
161            transaction_pool: transaction_pool.clone(),
162            keystore: keystore.clone(),
163            system_rpc_tx: system_rpc_tx.clone(),
164            impl_name: config.impl_name.clone(),
165            impl_version: config.impl_version.clone(),
166            chain_spec: config.chain_spec.as_ref(),
167            state_pruning: &config.state_pruning,
168            blocks_pruning: config.blocks_pruning,
169            backend: backend.clone(),
170            rpc_builder: &*rpc_builder,
171            metrics: None,
172            sync_oracle: sync_service.clone(),
173            tracing_execute_block: tracing_execute_block.clone(),
174        })
175    };
176
177    let rpc_api = gen_rpc_module()?;
178    let rpc_server_handle = start_rpc_servers(
179        &config.rpc,
180        config.prometheus_registry(),
181        &config.tokio_handle,
182        rpc_api,
183        rpc_runtime,
184        rpc_id_provider,
185    )?;
186
187    let listen_addrs = rpc_server_handle
188        .listen_addrs()
189        .iter()
190        .map(|socket_addr| {
191            let mut multiaddr: Multiaddr = socket_addr.ip().into();
192            multiaddr.push(Protocol::Tcp(socket_addr.port()));
193            multiaddr
194        })
195        .collect();
196
197    let in_memory_rpc = {
198        let mut module = gen_rpc_module()?;
199        module.extensions_mut().insert(DenyUnsafe::No);
200        module
201    };
202
203    let in_memory_rpc_handle = RpcHandlers::new(Arc::new(in_memory_rpc), listen_addrs);
204    // Spawn informant task
205    spawn_handle.spawn(
206        "informant",
207        None,
208        sc_informant::build(client.clone(), network, sync_service.clone()),
209    );
210
211    task_manager.keep_alive((config.base_path, rpc_server_handle));
212
213    Ok(in_memory_rpc_handle)
214}

216#[cfg(test)]
217mod tests {
218    use super::all_block_notifications;
219    use sc_transaction_pool::{TransactionPoolOptions, TransactionPoolType};
220
221    #[test]
222    fn single_state_pool_uses_best_block_notifications() {
223        assert!(!all_block_notifications(&TransactionPoolOptions::default()));
224    }
225
226    #[test]
227    fn fork_aware_pool_uses_all_block_notifications() {
228        let options = TransactionPoolOptions::new_with_params(
229            8192,
230            20 * 1024 * 1024,
231            None,
232            TransactionPoolType::ForkAware,
233            false,
234            false,
235        );
236        assert!(all_block_notifications(&options));
237    }
