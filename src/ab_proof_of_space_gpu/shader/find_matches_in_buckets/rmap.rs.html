1use crate::shader::constants::PARAM_BC;
2#[cfg(target_arch = "spirv")]
3use crate::shader::polyfills::ArrayIndexingPolyfill;
4use crate::shader::types::R;
5use spirv_std::arch::atomic_or;
6use spirv_std::memory::{Scope, Semantics};
7
8const PRESENCE_FLAGS_WORDS: usize = (PARAM_BC as usize * 2).div_ceil(u32::BITS as usize);
9
10#[derive(Debug, Copy, Clone)]
11pub(super) struct Rmap {
12    /// Store two bits per target, indicating whether matching elements are present (two at most)
13    presence_flags: [u32; PRESENCE_FLAGS_WORDS],
14}
15
16impl Rmap {
17    /// Returns `0`, `1` or `2` depending on whether `r` was present and whether there were
18    /// duplicates.
19    ///
20    /// `0` means not present, `1` means exactly one value. `2` means two or more values, but
21    /// doesn't track how many exactly.
22    pub(super) fn num_r_items(&self, r_target: R) -> u32 {
23        let bit_position = r_target.get() * 2;
24        let word_offset = (bit_position / u32::BITS) as usize;
25        let bit_offset = bit_position % u32::BITS;
26        // SAFETY: `bit_position` is within bounds of presence flags
27        let word = *unsafe { self.presence_flags.get_unchecked(word_offset) };
28
29        ((word >> bit_offset) & 0b11).count_ones()
30    }
31
32    pub(super) fn add_with_data_parallel(&mut self, r: R) {
33        let bit_position = r.get() * 2;
34        let word_offset = (bit_position / u32::BITS) as usize;
35        let bit_offset = bit_position % u32::BITS;
36        // SAFETY: `bit_position` is within bounds of presence flags
37        let word = unsafe { self.presence_flags.get_unchecked_mut(word_offset) };
38        let mask = 1 << bit_offset;
39
40        if cfg!(target_arch = "spirv") {
41            // SAFETY: TODO: Probably should not be unsafe to begin with:
42            //  https://github.com/Rust-GPU/rust-gpu/pull/394#issuecomment-3316594485
43            let prev_word_value = unsafe {
44                atomic_or::<_, { Scope::Workgroup as u32 }, { Semantics::NONE.bits() }>(word, mask)
45            };
46
47            if prev_word_value & mask != 0 {
48                // Bit was already set, so this is not the first such `r` value, set a flag
49                // indicating there was a duplicate
50                // SAFETY: TODO: Probably should not be unsafe to begin with:
51                //  https://github.com/Rust-GPU/rust-gpu/pull/394#issuecomment-3316594485
52                unsafe {
53                    atomic_or::<_, { Scope::Workgroup as u32 }, { Semantics::NONE.bits() }>(
54                        word,
55                        mask << 1u8,
56                    );
57                }
58            }
59        } else if *word & mask != 0 {
60            // Bit was already set, so this is not the first such `r` value, set a flag
61            // indicating there was a duplicate
62            *word |= mask << 1u8;
63        } else {
64            *word |= mask;
65        }
66    }
67}
