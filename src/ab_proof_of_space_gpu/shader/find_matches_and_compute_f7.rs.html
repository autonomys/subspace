45/// Upper-bound estimation of the number of matched elements per s-bucket.
46///
47/// Buckets are defined by the lower `NUM_S_BUCKETS.ilog2()` bits of the values. This is based on a
48/// Chernoff bound for the Poisson distribution with mean `lambda = 1`, ensuring the probability
49/// that any bucket exceeds the bound is less than `2^{-security_bits}`. The bound is
50/// `lambda + ceil(sqrt(3 * lambda * (NUM_S_BUCKETS.ilog2() + security_bits) * ln(2)))`.
51/// Accounts for the filter to values in `0..NUM_S_BUCKETS-1` by using the expected number of
52/// remaining elements ~`NUM_S_BUCKETS`, distributed uniformly across all `NUM_S_BUCKETS` buckets.
53const fn proofs_bucket_upper_bound(security_bits: u8) -> u64 {
54    // Lambda is the expected number of entries in a bucket:
55    // ~`NUM_S_BUCKETS / NUM_S_BUCKETS = 1`
56    const LAMBDA: u64 = 1;
57    // Approximation of ln(2) as a fraction: `ln(2) ≈ LN2_NUM / LN2_DEN`.
58    // This allows integer-only computation of the square root term involving ln(2).
59    const LN2_NUM: u128 = 693_147;
60    const LN2_DEN: u128 = 1_000_000;
61
62    // `log2(NUM_S_BUCKETS) + security_bits` for the union bound over `NUM_S_BUCKETS` buckets
63    let ks = NUM_S_BUCKETS.ilog2() as u128 + security_bits as u128;
64    // Compute numerator for the expression under the square root:
65    // `3 * lambda * ks * LN2_NUM`
66    let num = 3u128 * LAMBDA as u128 * ks * LN2_NUM;
67    // Denominator for ln(2): `LN2_DEN`
68    let den = LN2_DEN;
69
70    let ceil_div = num.div_ceil(den);
71
72    // Binary search to find the smallest `x` such that `x * x >= ceil_div`,
73    // which computes `ceil(sqrt(num / den))` without floating-point.
74    // We use a custom binary search over `u64` range because binary search in the standard library
75    // operates on sorted slices, not directly on integer ranges for solving inequalities like this.
76    let mut low = 0u64;
77    let mut high = u64::MAX;
78    while low < high {
79        let mid = low + (high - low) / 2;
80        let left = (mid as u128) * (mid as u128);
81        if left >= ceil_div {
82            high = mid;
83        } else {
84            low = mid + 1;
85        }
86    }
87    let add_term = low;
88
89    LAMBDA + add_term
90}

108/// Subspace's little-endian s-bucket convention (duplicated in the CPU `ab-proof-of-space` chiapos):
109/// maps a table-7 entry's `first_k_bits` to its s-bucket, returning a value `>= NUM_S_BUCKETS` for
110/// entries whose low `LOW_BITS` bits are set (the caller discards those). Consensus verification
111/// derives the challenge from the s-bucket with the same little-endian byte layout, so this
112/// convention is fixed.
113#[inline(always)]
114fn little_endian_s_bucket(first_k_bits: u32) -> u32 {
115    if first_k_bits.trailing_zeros() < LOW_BITS {
116        return u32::MAX;
117    }
118    let s_bucket = (first_k_bits >> LOW_BITS) & 0xffff;
119    (s_bucket & 0xff) << 8 | s_bucket >> 8
120}

122/// # Safety
123/// `bucket_index` must be within range `0..REDUCED_MATCHES_COUNT`. `matches_count` elements in
124/// `matches` must be initialized, `matches` must have valid pointers into left/right buckets and
125/// `parent_metadatas`.
126#[inline(always)]
127#[expect(
128    clippy::too_many_arguments,
129    reason = "Both I/O and Vulkan stuff together take a lot of arguments"
130)]
131unsafe fn compute_f7_into_buckets_inner(
132    index: u32,
133    left_bucket_base: u32,
134    absolute_position_base: u32,
135    left_bucket: &[PositionR; MAX_BUCKET_SIZE],
136    // TODO: `&[Match]` would have been nicer, but it currently doesn't compile:
137    //  https://github.com/Rust-GPU/rust-gpu/issues/241#issuecomment-3005693043
138    matches: &[MaybeUninit<Match>; MAX_BUCKET_SIZE],
139    // TODO: This should have been `&[[Metadata; REDUCED_MATCHES_COUNT]; NUM_MATCH_BUCKETS]`, but
140    //  it currently doesn't compile if flattened:
141    //  https://github.com/Rust-GPU/rust-gpu/issues/241#issuecomment-3005693043
142    parent_metadatas: &[Metadata; REDUCED_MATCHES_COUNT * NUM_MATCH_BUCKETS],
143    table_6_proof_targets_sizes: &mut [u32; NUM_S_BUCKETS],
144    table_6_proof_targets: &mut [[MaybeUninit<ProofTargets>; NUM_ELEMENTS_PER_S_BUCKET];
145             NUM_S_BUCKETS],
146    bucket_scratch: &[PositionR; REDUCED_BUCKET_SIZE],
147) {
148    // SAFETY: Guaranteed by function contract
149    let (bucket_offset, r_target, positions_offset) =
150        unsafe { matches.get_unchecked(index as usize).assume_init() }.split();
151
152    // SAFETY: Guaranteed by function contract
153    let left_position_r = *unsafe { left_bucket.get_unchecked(bucket_offset as usize) };
154    let left_position = left_position_r.position;
155    let left_r = left_position_r.r.get();
156
157    // Repurpose variable for two purposes to save on registers
158    let mut right_position_or_skip = positions_offset;
159    // TODO: More idiomatic version currently doesn't compile:
160    //  https://github.com/Rust-GPU/rust-gpu/issues/241#issuecomment-3005693043
161    #[expect(
162        clippy::needless_range_loop,
163        reason = "Intentional workaround for rust-gpu"
164    )]
165    for offset in 0..REDUCED_BUCKET_SIZE {
166        let position_r = bucket_scratch[offset];
167        if position_r.r.get() == r_target {
168            if right_position_or_skip == 0 {
169                right_position_or_skip = position_r.position;
170                break;
171            }
172
173            right_position_or_skip -= 1;
174        }
175    }
176    let right_position = right_position_or_skip;
177
178    // TODO: Correct version currently doesn't compile:
179    //  https://github.com/Rust-GPU/rust-gpu/issues/241#issuecomment-3005693043
180    // let left_metadata = parent_metadatas[usize::from(left_position)];
181    // let right_metadata = parent_metadatas[usize::from(right_position)];
182    // SAFETY: Guaranteed by function contract
183    let left_metadata = *unsafe { parent_metadatas.get_unchecked(left_position as usize) };
184    // SAFETY: Guaranteed by function contract
185    let right_metadata = *unsafe { parent_metadatas.get_unchecked(right_position as usize) };
186
187    let (y, _) = compute_fn_impl::<TABLE_NUMBER, PARENT_TABLE_NUMBER>(
188        Y::from(left_bucket_base + left_r),
189        left_metadata,
190        right_metadata,
191    );
192
193    let s_bucket = little_endian_s_bucket(y.first_k_bits()) as usize;
194    // TODO: More idiomatic version currently doesn't compile:
195    //  https://github.com/Rust-GPU/rust-gpu/issues/241#issuecomment-3005693043
196    // let Some(bucket_count) = bucket_sizes.get_mut(s_bucket) else {
197    //     continue;
198    // };
199    if s_bucket >= NUM_S_BUCKETS {
200        return;
201    }
202    let bucket_size = &mut table_6_proof_targets_sizes[s_bucket];
203    // SAFETY: TODO: Probably should not be unsafe to begin with:
204    //  https://github.com/Rust-GPU/rust-gpu/pull/394#issuecomment-3316594485
205    let bucket_offset = unsafe {
206        atomic_i_increment::<_, { Scope::QueueFamily as u32 }, { Semantics::NONE.bits() }>(
207            bucket_size,
208        )
209    };
210
211    // TODO: Maybe store `absolute_position` separately from more densely populated positions,
212    //  then per-s-bucket `atomic_u_min()` can be done to store the minimum pointer. This all
213    //  will use less memory overall
214    // SAFETY: `s_bucket` is checked above to be correct. Bucket size upper bound is known
215    // statically to be [`NUM_ELEMENTS_PER_S_BUCKET`], so `bucket_offset` is also always within
216    // bounds.
217    unsafe {
218        table_6_proof_targets
219            .get_unchecked_mut(s_bucket)
220            .get_unchecked_mut(bucket_offset as usize)
221    }
222    .write(ProofTargets {
223        absolute_position: absolute_position_base + index,
224        positions: [left_position, right_position],
225    });
226}

228/// # Safety
229/// `bucket_index` must be within range `0..REDUCED_MATCHES_COUNT`. `matches_count` elements in
230/// `matches` must be initialized, `matches` must have valid pointers into left/right buckets and
231/// `parent_metadatas`.
232#[inline(always)]
233#[expect(
234    clippy::too_many_arguments,
235    reason = "Both I/O and Vulkan stuff together take a lot of arguments"
236)]
237unsafe fn compute_f7_into_buckets(
238    local_invocation_id: u32,
239    left_bucket_index: u32,
240    left_bucket: &[PositionR; MAX_BUCKET_SIZE],
241    right_bucket: &[PositionR; MAX_BUCKET_SIZE],
242    matches_count: usize,
243    // TODO: `&[Match]` would have been nicer, but it currently doesn't compile:
244    //  https://github.com/Rust-GPU/rust-gpu/issues/241#issuecomment-3005693043
245    matches: &[MaybeUninit<Match>; MAX_BUCKET_SIZE],
246    // TODO: This should have been `&[[Metadata; REDUCED_MATCHES_COUNT]; NUM_MATCH_BUCKETS]`, but
247    //  it currently doesn't compile if flattened:
248    //  https://github.com/Rust-GPU/rust-gpu/issues/241#issuecomment-3005693043
249    parent_metadatas: &[Metadata; REDUCED_MATCHES_COUNT * NUM_MATCH_BUCKETS],
250    table_6_proof_targets_sizes: &mut [u32; NUM_S_BUCKETS],
251    table_6_proof_targets: &mut [[MaybeUninit<ProofTargets>; NUM_ELEMENTS_PER_S_BUCKET];
252             NUM_S_BUCKETS],
253    bucket_scratch: &mut [PositionR; REDUCED_BUCKET_SIZE],
254) {
255    // Load the right bucket into shared memory for faster access
256    for bucket_offset in
257        (local_invocation_id as usize..REDUCED_BUCKET_SIZE).step_by(WORKGROUP_SIZE as usize)
258    {
259        bucket_scratch[bucket_offset] = right_bucket[bucket_offset];
260    }
261
262    workgroup_memory_barrier_with_group_sync();
263
264    let left_bucket_base = left_bucket_index * u32::from(PARAM_BC);
265    let absolute_position_base = left_bucket_index * REDUCED_MATCHES_COUNT as u32;
266
267    const {
268        assert!(MAX_BUCKET_SIZE == WORKGROUP_SIZE as usize * 2);
269    }
270    // TODO: This should have been a loop, but register usage is too high, see:
271    //  https://github.com/Rust-GPU/rust-gpu/issues/462
272    // SAFETY: Guaranteed by function contract
273    unsafe {
274        if (local_invocation_id as usize) < matches_count {
275            compute_f7_into_buckets_inner(
276                local_invocation_id,
277                left_bucket_base,
278                absolute_position_base,
279                left_bucket,
280                matches,
281                parent_metadatas,
282                table_6_proof_targets_sizes,
283                table_6_proof_targets,
284                bucket_scratch,
285            );
286        }
287        if ((local_invocation_id + WORKGROUP_SIZE) as usize) < matches_count {
288            compute_f7_into_buckets_inner(
289                local_invocation_id + WORKGROUP_SIZE,
290                left_bucket_base,
291                absolute_position_base,
292                left_bucket,
293                matches,
294                parent_metadatas,
295                table_6_proof_targets_sizes,
296                table_6_proof_targets,
297                bucket_scratch,
298            );
299        }
300    }
301}

303/// This is similar to `find_matches_and_compute_fn`, but it stores results in buckets grouped by
304/// s-buckets, which is how proofs can later be found efficiently.
305///
306/// Buckets need to be sorted by position afterward due to concurrent writes that do not have
307/// deterministic order. Content of the bucket beyond the size specified in `bucket_sizes` is
308/// undefined.
309///
310/// # Safety
311/// Must be called from [`WORKGROUP_SIZE`] threads.  All buckets must contain valid positions.
312#[spirv(compute(threads(256), entry_point_name = "find_matches_and_compute_f7"))]
313#[expect(
314    clippy::too_many_arguments,
315    reason = "Both I/O and Vulkan stuff together take a lot of arguments"
316)]
317pub unsafe fn find_matches_and_compute_f7(
318    #[spirv(local_invocation_id)] local_invocation_id: UVec3,
319    #[spirv(workgroup_id)] workgroup_id: UVec3,
320    #[spirv(storage_buffer, descriptor_set = 0, binding = 0)] parent_buckets: &[[PositionR; MAX_BUCKET_SIZE];
321         NUM_BUCKETS],
322    #[spirv(storage_buffer, descriptor_set = 0, binding = 1)]
323    parent_metadatas: &[Metadata; REDUCED_MATCHES_COUNT * NUM_MATCH_BUCKETS],
324    #[spirv(storage_buffer, descriptor_set = 0, binding = 2)]
325    table_6_proof_targets_sizes: &mut [u32; NUM_S_BUCKETS],
326    #[spirv(storage_buffer, descriptor_set = 0, binding = 3)]
327    table_6_proof_targets: &mut [[MaybeUninit<ProofTargets>; NUM_ELEMENTS_PER_S_BUCKET];
328             NUM_S_BUCKETS],
329    #[spirv(workgroup)] matches: &mut [MaybeUninit<Match>; MAX_BUCKET_SIZE],
330    #[spirv(workgroup)] shared: &mut FindMatchesAndComputeF7Shared,
331) {
332    let local_invocation_id = local_invocation_id.x;
333    let workgroup_id = workgroup_id.x;
334
335    let left_bucket_index = workgroup_id as usize;
336    let left_bucket = &parent_buckets[left_bucket_index];
337    let right_bucket = &parent_buckets[left_bucket_index + 1];
338    let left_bucket_index = left_bucket_index as u32;
339
340    // TODO: Truncate buckets to reduced size here once it compiles:
341    //  https://github.com/Rust-GPU/rust-gpu/issues/241#issuecomment-3005693043
342    // SAFETY: Guaranteed by function contract
343    let matches_count = unsafe {
344        find_matches_in_buckets_impl(
345            local_invocation_id,
346            left_bucket_index,
347            left_bucket,
348            right_bucket,
349            matches,
350            &mut shared.find_matches_shared,
351        )
352    };
353
354    // SAFETY: Guaranteed by function contract and call to `find_matches_in_buckets_impl`
355    unsafe {
356        compute_f7_into_buckets(
357            local_invocation_id,
358            left_bucket_index,
359            left_bucket,
360            right_bucket,
361            matches_count as usize,
362            matches,
363            parent_metadatas,
364            table_6_proof_targets_sizes,
365            table_6_proof_targets,
366            &mut shared.bucket_scratch,
367        );
368    }
