41impl Y {

44    pub(in super::super) const fn first_k_bits(self) -> u32 {
45        self.0 >> PARAM_EXT
46    }

48    pub(super) fn into_bucket_index_and_r(self) -> (u32, R) {
49        let bucket_index = self.0 / u32::from(PARAM_BC);
50        let r = self.0 % u32::from(PARAM_BC);
51        // SAFETY: `r` is within `0..PARAM_BC` range
52        let r = unsafe { R::new(r) };
53        (bucket_index, r)
54    }

180#[derive(Copy, Clone, Eq, PartialEq)]
181#[repr(C)]
182pub struct Match(u32);
183
184impl fmt::Debug for Match {
185    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
186        let (bucket_offset, r_target, positions_offset) = self.split();
187        f.debug_struct("Match")
188            .field("bucket_offset", &bucket_offset)
189            .field("r_target", &r_target)
190            .field("positions_offset", &positions_offset)
191            .finish()
192    }
193}
194
195impl Match {
196    /// Match that can't exist
197    pub(super) const SENTINEL: Self = Self(u32::MAX);
198
199    /// NOTE: `m` is only present here to ensure correct sorting order, it is not used for anything
200    /// else.
201    ///
202    /// # Safety
203    /// `bucket_offset` value must be within `0..MAX_BUCKET_SIZE` range, `m` must be within
204    /// `0..PARAM_M` range and `r_target` must be within `0..PARAM_BC` range
205    #[inline(always)]
206    pub(super) unsafe fn new(bucket_offset: u32, m: u32, r_target: u32) -> Self {
207        #[expect(clippy::int_plus_one, reason = "Better explains the underlying logic")]
208        const {
209            assert!(
210                (MAX_BUCKET_SIZE - 1).bit_width() + (PARAM_M - 1).bit_width() + R::BITS + 1
211                    <= u32::BITS
212            );
213        }
214
215        // TODO: `const {}` is a workaround for https://github.com/Rust-GPU/rust-gpu/issues/322 and
216        //  shouldn't be necessary otherwise
217        Self(
218            (bucket_offset << const { (PARAM_M - 1).bit_width() + R::BITS + 1 })
219                | (m << const { R::BITS + 1 })
220                | (r_target << 1),
221        )
222    }
223
224    /// Initially, `Match` assumes the first position is used. This allows changing it to the
225    /// second.
226    ///
227    /// # Safety
228    /// Must only be called once on any `Match` instance.
229    #[inline(always)]
230    pub(super) fn second_second_position(mut self) -> Self {
231        self.0 += 1;
232        self
233    }
234
235    /// Returns `(bucket_offset, r_target, positions_offset)`
236    #[inline(always)]
237    pub(super) fn split(self) -> (u32, u32, u32) {
238        // TODO: `const {}` is a workaround for https://github.com/Rust-GPU/rust-gpu/issues/322 and
239        //  shouldn't be necessary otherwise
240        (
241            self.0 >> const { (PARAM_M - 1).bit_width() + R::BITS + 1 },
242            (self.0 >> 1) & const { u32::MAX >> (u32::BITS - R::BITS) },
243            self.0 & 1,
244        )
245    }
246
247    /// Extracts a key that can be used for sorting matches
248    #[inline(always)]
249    pub(super) fn cmp_key(self) -> u32 {
250        self.0
251    }
