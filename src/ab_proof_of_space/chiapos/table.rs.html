160/// Compute `y`s of the first table out of the ChaCha8 keystream
161#[cfg(feature = "alloc")]
162#[cfg_attr(feature = "no-panic", no_panic::no_panic)]
163fn compute_table_1_ys<'a, const K: u8>(
164    partial_ys: &[u8; TABLE_1_PARTIAL_YS_SIZE::<K>],
165    ys: &'a mut [MaybeUninit<Y>; MAX_TABLE_SIZE::<K>],
166) -> &'a [Y] {
167    for ((ys, xs_batch_start), partial_ys) in ys
168        .as_chunks_mut::<COMPUTE_F1_SIMD_FACTOR>()
169        .0
170        .iter_mut()
171        .zip((X::ZERO..).step_by(COMPUTE_F1_SIMD_FACTOR))
172        .zip(partial_ys.as_chunks::<{ TABLE_1_YS_BATCH_SIMD::<K> }>().0)
173    {
174        let xs =
175            Simd::splat(u32::from(xs_batch_start)) + Simd::from_array(array::from_fn(|i| i as u32));
176        let ys_batch = compute_f1_simd::<K>(xs, partial_ys);
177
178        ys.write_copy_of_slice(&ys_batch);
179    }
180
181    // SAFETY: The keystream covers the whole table, so all elements were initialized
182    unsafe { ys.assume_init_ref() }
183}

185/// Calculate a probabilistic upper bound on the Chia bucket size for a given `k` and
186/// `security_bits` (security level).
187///
188/// This is based on a Chernoff bound for the Poisson distribution with mean
189/// `lambda = PARAM_BC / 2^PARAM_EXT`, ensuring the probability that any bucket exceeds the bound is
190/// less than `2^{-security_bits}`.
191/// The bound is lambda + ceil(sqrt(3 * lambda * (k + security_bits) * ln(2))).
192#[cfg(feature = "alloc")]
193const fn bucket_size_upper_bound(k: u8, security_bits: u8) -> usize {
194    // Lambda is the expected number of entries in a bucket, approximated as
195    // `PARAM_BC / 2^PARAM_EXT`. It is independent of `k`.
196    const LAMBDA: u64 = PARAM_BC as u64 / 2u64.pow(PARAM_EXT as u32);
197    // Approximation of ln(2) as a fraction: ln(2) ≈ LN2_NUM / LN2_DEN.
198    // This allows integer-only computation of the square root term involving ln(2).
199    const LN2_NUM: u128 = 693_147;
200    const LN2_DEN: u128 = 1_000_000;
201
202    // `k + security_bits` for the union bound over ~2^k intervals
203    let ks = k as u128 + security_bits as u128;
204    // Compute numerator for the expression under the square root:
205    // `3 * lambda * (k + security_bits) * LN2_NUM`
206    let num = 3u128 * LAMBDA as u128 * ks * LN2_NUM;
207    // Denominator for ln(2): `LN2_DEN`
208    let den = LN2_DEN;
209
210    let ceil_div = num.div_ceil(den);
211
212    // Binary search to find the smallest `x` such that `x * x * den >= num`,
213    // which computes `ceil(sqrt(num / den))` without floating-point.
214    // We use a custom binary search over `u64` range because binary search in the standard library
215    // operates on sorted slices, not directly on integer ranges for solving inequalities like this.
216    let mut low = 0u64;
217    let mut high = u64::MAX;
218    while low < high {
219        let mid = low + (high - low) / 2;
220        let left = (mid as u128) * (mid as u128);
221        if left >= ceil_div {
222            high = mid;
223        } else {
224            low = mid + 1;
225        }
226    }
227    let add_term = low;
228
229    (LAMBDA + add_term) as usize
230}

248/// The part of [`group_by_buckets()`] that works with already allocated memory
249#[cfg(feature = "alloc")]
250#[cfg_attr(feature = "no-panic", no_panic::no_panic)]
251fn group_by_buckets_internal<const K: u8>(
252    ys: &[Y],
253    buckets: &mut [[MaybeUninit<(Position, Y)>; REDUCED_BUCKET_SIZE]; NUM_BUCKETS::<K>],
254) {
255    let mut bucket_lengths = [0_u16; NUM_BUCKETS::<K>];
256
257    for (&y, position) in ys.iter().zip(Position::ZERO..) {
258        let bucket_index = (u32::from(y) / u32::from(PARAM_BC)) as usize;
259
260        // SAFETY: Bucket is obtained using division by `PARAM_BC` and fits by definition
261        unsafe {
262            hint::assert_unchecked(bucket_index < NUM_BUCKETS::<K>);
263        }
264        let bucket_length = bucket_lengths
265            .get_mut(bucket_index)
266            .expect("Bucket index is within bounds as asserted above; qed");
267        let bucket = buckets
268            .get_mut(bucket_index)
269            .expect("Bucket index is within bounds as asserted above; qed");
270
271        // Entries past `REDUCED_BUCKET_SIZE` are thrown away
272        if let Some(entry) = bucket.get_mut(usize::from(*bucket_length)) {
273            entry.write((position, y));
274            *bucket_length += 1;
275        }
276    }
277
278    // SAFETY: Bucket lengths are limited to `REDUCED_BUCKET_SIZE` above
279    unsafe {
280        fill_bucket_tails::<K>(buckets, &bucket_lengths);
281    }
282}

305/// Count how many `y`s of a single chunk belong to each bucket.
306///
307/// # Safety
308/// `counts` must be the number of initialized `y`s in each entry of `ys`.
309#[cfg(feature = "parallel")]
310#[cfg_attr(feature = "no-panic", no_panic::no_panic)]
311unsafe fn count_ys_in_buckets<const K: u8>(
312    ys: &[[MaybeUninit<Y>; REDUCED_MATCHES_COUNT]],
313    counts: &[u16],
314    chunk_offsets: &mut [u16; NUM_BUCKETS::<K>],
315) {
316    for (ys, &count) in ys.iter().zip(counts) {
317        // SAFETY: Function contract guarantees that this many `y`s are initialized, hence the
318        // count is also within bounds
319        let ys = unsafe { ys.get_unchecked(..usize::from(count)).assume_init_ref() };
320
321        for &y in ys {
322            let bucket_index = (u32::from(y) / u32::from(PARAM_BC)) as usize;
323
324            // SAFETY: Bucket is obtained using division by `PARAM_BC` and fits by definition
325            unsafe {
326                *chunk_offsets.get_unchecked_mut(bucket_index) += 1;
327            }
328        }
329    }

332/// Write `y`s of a single chunk into the buckets they belong to, starting at the offsets reserved
333/// for this chunk.
334///
335/// # Safety
336/// `counts` must be the number of initialized `y`s in each entry of `ys`, `batch_start` must be
337/// the position of the first `y` of `ys`, `chunk_offsets` must be the offsets within buckets that
338/// are reserved exclusively for this chunk.
339#[cfg(feature = "parallel")]
340#[cfg_attr(feature = "no-panic", no_panic::no_panic)]
341#[expect(clippy::type_complexity, reason = "Internal API")]
342unsafe fn scatter_ys_into_buckets<const K: u8>(
343    ys: &[[MaybeUninit<Y>; REDUCED_MATCHES_COUNT]],
344    counts: &[u16],
345    chunk_offsets: &[u16; NUM_BUCKETS::<K>],
346    batch_start: Position,
347    buckets: &[[SyncUnsafeCell<MaybeUninit<(Position, Y)>>; REDUCED_BUCKET_SIZE]; NUM_BUCKETS::<K>],
348) {
349    let mut bucket_offsets = *chunk_offsets;
350
351    for ((ys, &count), batch_start) in ys
352        .iter()
353        .zip(counts)
354        .zip((batch_start..).step_by(REDUCED_MATCHES_COUNT))
355    {
356        // SAFETY: Function contract guarantees that this many `y`s are initialized, hence the
357        // count is also within bounds
358        let ys = unsafe { ys.get_unchecked(..usize::from(count)).assume_init_ref() };
359
360        for (&y, position) in ys.iter().zip(batch_start..) {
361            let bucket_index = (u32::from(y) / u32::from(PARAM_BC)) as usize;
362
363            // SAFETY: Bucket is obtained using division by `PARAM_BC` and fits by definition
364            unsafe {
365                hint::assert_unchecked(bucket_index < NUM_BUCKETS::<K>);
366            }
367            let bucket_offset = bucket_offsets
368                .get_mut(bucket_index)
369                .expect("Bucket index is within bounds as asserted above; qed");
370            let bucket = buckets
371                .get(bucket_index)
372                .expect("Bucket index is within bounds as asserted above; qed");
373
374            // Entries past `REDUCED_BUCKET_SIZE` are thrown away
375            if let Some(entry) = bucket.get(usize::from(*bucket_offset)) {
376                // SAFETY: Function contract guarantees that offsets are exclusive to this chunk,
377                // so this is the only place where this entry is accessed
378                unsafe { &mut *entry.get() }.write((position, y));
379                *bucket_offset += 1;
380            }
381        }
382    }
383}

385/// Similar to [`group_by_buckets()`], but processes buckets instead of a flat list of `y`s.
386///
387/// Done as a counting sort: the first pass counts how many `y`s each chunk of the input
388/// contributes to each bucket, which after a prefix sum tells every chunk where in each bucket its
389/// `y`s go, so that the second (and by far the most expensive) pass can scatter them in parallel.
390///
391/// Both passes use parallel iterators rather than `rayon::broadcast()` that the rest of this
392/// module prefers. A broadcast has to reach every thread of the pool before it makes progress, and
393/// these two are called six times per set of tables while other sets are typically being built on
394/// the same pool, at which point waiting for the whole pool costs several times more than the
395/// grouping itself.
396///
397/// # Safety
398/// `counts` must be the number of initialized `y`s in each entry of `ys`.
399#[cfg(feature = "parallel")]
400unsafe fn group_by_buckets_from_buckets<const K: u8>(
401    ys: &[[MaybeUninit<Y>; REDUCED_MATCHES_COUNT]; NUM_BUCKET_PAIRS::<K>],
402    counts: &[u16; NUM_BUCKET_PAIRS::<K>],
403) -> Box<[[(Position, Y); REDUCED_BUCKET_SIZE]; NUM_BUCKETS::<K>]> {
404    let chunk_size = GROUP_BY_BUCKETS_CHUNK_SIZE::<K>;
405    // The last chunk is smaller unless bucket pairs divide evenly
406    let chunk_range = |chunk_index: usize| {
407        let start = chunk_index * chunk_size;
408
409        start..(start + chunk_size).min(NUM_BUCKET_PAIRS::<K>)
410    };
411
412    // A chunk only ever counts a subset of a bucket, and a bucket holds at most
413    // `MAX_BUCKET_SIZE` entries, so counts and the offsets they turn into both fit `u16`
414    // SAFETY: Zeroes are the correct initial counts
415    let mut chunk_offsets = unsafe {
416        Box::<[[u16; NUM_BUCKETS::<K>]; GROUP_BY_BUCKETS_CHUNKS::<K>]>::new_zeroed().assume_init()
417    };
418
419    chunk_offsets
420        .par_iter_mut()
421        .enumerate()
422        .for_each(|(chunk_index, chunk_offsets)| {
423            let chunk_range = chunk_range(chunk_index);
424            let (ys, counts) = (&ys[chunk_range.clone()], &counts[chunk_range]);
425
426            // SAFETY: Guaranteed by function contract
427            unsafe {
428                count_ys_in_buckets::<K>(ys, counts, chunk_offsets);
429            }
430        });
431
432    // Turn counts into offsets, ending up with the number of `y`s in each bucket
433    let mut bucket_lengths = [0_u16; NUM_BUCKETS::<K>];
434    for chunk_offsets in &mut *chunk_offsets {
435        for (chunk_offset, bucket_length) in chunk_offsets.iter_mut().zip(&mut bucket_lengths) {
436            let offset = *bucket_length;
437            // Buckets are limited to `REDUCED_BUCKET_SIZE`, anything past that is thrown away,
438            // which also keeps offsets within `u16`
439            *bucket_length = (*bucket_length + *chunk_offset).min(REDUCED_BUCKET_SIZE as u16);
440            *chunk_offset = offset;
441        }
442    }
443
444    // Each element is wrapped individually because chunks that share a bucket write different
445    // elements of it concurrently, and a `&mut` to the whole bucket would claim unique access to
446    // elements belonging to other chunks
447    // SAFETY: Contents is `MaybeUninit`
448    let buckets = unsafe { Box::<_>::new_uninit().assume_init() };
449
450    chunk_offsets
451        .par_iter()
452        .enumerate()
453        .for_each(|(chunk_index, chunk_offsets)| {
454            let chunk_range = chunk_range(chunk_index);
455            let batch_start = Position::from((chunk_range.start * REDUCED_MATCHES_COUNT) as u32);
456            let (ys, counts) = (&ys[chunk_range.clone()], &counts[chunk_range]);
457
458            // SAFETY: Guaranteed by function contract, offsets of different chunks were made
459            // disjoint by the prefix sum above
460            unsafe {
461                scatter_ys_into_buckets::<K>(ys, counts, chunk_offsets, batch_start, &buckets);
462            }
463        });
464
465    let mut buckets = strip_sync_unsafe_cell_elements(buckets);
466
467    // SAFETY: Bucket lengths were limited to `REDUCED_BUCKET_SIZE` by the prefix sum above
468    unsafe {
469        fill_bucket_tails::<K>(&mut buckets, &bucket_lengths);
470    }
471
472    // SAFETY: All entries are initialized
473    unsafe { Box::from_raw(Box::into_raw(buckets).cast()) }
474}

484/// `partial_y_offset` is in bits within `partial_y`
485#[cfg_attr(feature = "no-panic", no_panic::no_panic)]
486pub(super) fn compute_f1<const K: u8>(x: X, seed: &Seed) -> Y {
487    const U32S_PER_BLOCK: usize = size_of::<ChaCha8Block>() / size_of::<u32>();
488
489    let skip_bits = u32::from(K) * u32::from(x);
490    let skip_u32s = skip_bits / u32::BITS;
491    let partial_y_offset = skip_bits % u32::BITS;
492
493    let initial_state = ChaCha8State::init(seed, &[0; _]);
494    let first_block_counter = skip_u32s / U32S_PER_BLOCK as u32;
495    let u32_in_first_block = skip_u32s as usize % U32S_PER_BLOCK;
496
497    let first_block = initial_state.compute_block(first_block_counter);
498    let hi = first_block[u32_in_first_block].to_be();
499
500    // TODO: Is SIMD version of `compute_block()` that produces two blocks at once possible?
501    let lo = if u32_in_first_block + 1 == U32S_PER_BLOCK {
502        // Spilled over into the second block
503        let second_block = initial_state.compute_block(first_block_counter + 1);
504        second_block[0].to_be()
505    } else {
506        first_block[u32_in_first_block + 1].to_be()
507    };
508
509    // `K` bits of `partial_y` followed by `PARAM_EXT` extra bits that will be cleared
510    let pre_y = hi.funnel_shl(lo, partial_y_offset) >> (u32::BITS - u32::from(K + PARAM_EXT));
511    // Mask for clearing the rest of bits of `pre_y`.
512    let pre_y_mask = u32::MAX << PARAM_EXT;
513
514    // Extract `PARAM_EXT` most significant bits from `x` and store in the final offset of
515    // eventual `y` with the rest of bits being zero (`x` is `0..2^K`)
516    let pre_ext = u32::from(x) >> (K - PARAM_EXT);
517
518    // Combine all of the bits together:
519    // [padding zero bits][`K` bits rom `partial_y`][`PARAM_EXT` bits from `x`]
520    Y::from((pre_y & pre_y_mask) | pre_ext)
521}

523#[cfg(any(feature = "alloc", test))]
524#[cfg_attr(feature = "no-panic", no_panic::no_panic)]
525pub(super) fn compute_f1_simd<const K: u8>(
526    xs: Simd<u32, COMPUTE_F1_SIMD_FACTOR>,
527    partial_ys: &[u8; TABLE_1_YS_BATCH_SIMD::<K>],
528) -> [Y; COMPUTE_F1_SIMD_FACTOR] {
529    // Each element contains `K` desired bits of `partial_ys` in the final offset of eventual `ys`
530    // with the rest of bits being in undefined state
531    let pre_ys_bytes = array::from_fn(|i| {
532        let partial_y_offset = i * usize::from(K);
533        let partial_y_length =
534            (partial_y_offset % u8::BITS as usize + usize::from(K)).div_ceil(u8::BITS as usize);
535        let mut pre_y_bytes = 0u64.to_be_bytes();
536        pre_y_bytes[..partial_y_length].copy_from_slice(
537            &partial_ys[partial_y_offset / u8::BITS as usize..][..partial_y_length],
538        );
539
540        u64::from_be_bytes(pre_y_bytes)
541    });
542    let pre_ys_right_offset = array::from_fn(|i| {
543        let partial_y_offset = i as u32 * u32::from(K);
544        u64::from(u64::BITS - u32::from(K + PARAM_EXT) - partial_y_offset % u8::BITS)
545    });
546    let pre_ys = Simd::from_array(pre_ys_bytes) >> Simd::from_array(pre_ys_right_offset);
547
548    // Mask for clearing the rest of bits of `pre_ys`.
549    let pre_ys_mask = Simd::splat(
550        (u32::MAX << usize::from(PARAM_EXT))
551            & (u32::MAX >> (u32::BITS as usize - usize::from(K + PARAM_EXT))),
552    );
553
554    // Extract `PARAM_EXT` most significant bits from `xs` and store in the final offset of
555    // eventual `ys` with the rest of bits being in undefined state.
556    let pre_exts = xs >> Simd::splat(u32::from(K - PARAM_EXT));
557
558    // Combine all of the bits together:
559    // [padding zero bits][`K` bits rom `partial_y`][`PARAM_EXT` bits from `x`]
560    let ys = (pre_ys.cast() & pre_ys_mask) | pre_exts;
561
562    Y::array_from_repr(ys.to_array())
563}

565/// For verification use [`has_match`] instead.
566///
567/// # Safety
568/// Left and right bucket positions must correspond to the parent table.
569// TODO: Try to reduce the `matches` size further by processing `left_bucket` in chunks (like halves
570//  for example)
571#[cfg(feature = "alloc")]
572#[cfg_attr(feature = "no-panic", no_panic::no_panic)]
573unsafe fn find_matches_in_buckets<'a>(
574    left_bucket_index: u32,
575    left_bucket: &[(Position, Y); REDUCED_BUCKET_SIZE],
576    right_bucket: &[(Position, Y); REDUCED_BUCKET_SIZE],
577    // `PARAM_M * 2` corresponds to the upper bound number of matches a single `y` in the
578    // left bucket might have here
579    matches: &'a mut [MaybeUninit<Match>; REDUCED_MATCHES_COUNT + usize::from(PARAM_M) * 2],
580) -> &'a [Match] {
581    let left_base = left_bucket_index * u32::from(PARAM_BC);
582    let right_base = left_base + u32::from(PARAM_BC);
583
584    let mut rmap = Rmap::new();
585    for &(right_position, y) in right_bucket {
586        if right_position == Position::SENTINEL {
587            break;
588        }
589        let r = R::from((u32::from(y) - right_base) as u16);
590        // SAFETY: `r` is within `0..PARAM_BC` range by definition, the right bucket is limited to
591        // `REDUCED_BUCKETS_SIZE`
592        unsafe {
593            rmap.add(r, right_position);
594        }
595    }
596
597    let left_targets = LeftTargets::new(left_base % 2 == 1);
598    let mut next_match_index = 0;
599
600    for &(left_position, y) in left_bucket {
601        // `next_match_index >= REDUCED_MATCHES_COUNT` is crucial to make sure
602        if left_position == Position::SENTINEL || next_match_index >= REDUCED_MATCHES_COUNT {
603            // Sentinel values are padded to the end of the bucket
604            break;
605        }
606
607        let r = R::from((u32::from(y) - left_base) as u16);
608
609        for r_target in left_targets.calculate(r) {
610            // SAFETY: Targets are always limited to `PARAM_BC`
611            let [right_position_a, right_position_b] = unsafe { rmap.get(r_target) };
612
613            if right_position_a != Position::SENTINEL {
614                // SAFETY: Iteration will stop before `REDUCED_MATCHES_COUNT + PARAM_M * 2`
615                // elements is inserted
616                unsafe { matches.get_unchecked_mut(next_match_index) }.write(Match {
617                    left_position,
618                    left_y: y,
619                    right_position: right_position_a,
620                });
621                next_match_index += 1;
622
623                if right_position_b != Position::SENTINEL {
624                    // SAFETY: Iteration will stop before
625                    // `REDUCED_MATCHES_COUNT + PARAM_M * 2` elements is inserted
626                    unsafe { matches.get_unchecked_mut(next_match_index) }.write(Match {
627                        left_position,
628                        left_y: y,
629                        right_position: right_position_b,
630                    });
631                    next_match_index += 1;
632                }
633            }
634        }
635    }
636
637    // SAFETY: Initialized this many matches, which is also why the number of matches is within
638    // bounds
639    unsafe { matches.get_unchecked(..next_match_index).assume_init_ref() }
640}

642/// Simplified version of [`find_matches_in_buckets`] for verification purposes.
643#[cfg_attr(feature = "no-panic", no_panic::no_panic)]
644pub(super) fn has_match(left_y: Y, right_y: Y) -> bool {
645    let left_bucket_index = u32::from(left_y) / u32::from(PARAM_BC);
646    let left_r = R::from((u32::from(left_y) - left_bucket_index * u32::from(PARAM_BC)) as u16);
647    let right_r = R::from((u32::from(right_y) % u32::from(PARAM_BC)) as u16);
648
649    LeftTargets::new(left_bucket_index % 2 == 1).contains(left_r, right_r)
650}

652#[inline(always)]
653#[cfg_attr(feature = "no-panic", no_panic::no_panic)]
654pub(super) fn compute_fn<const K: u8, const TABLE_NUMBER: u8, const PARENT_TABLE_NUMBER: u8>(
655    y: Y,
656    left_metadata: Metadata<K, PARENT_TABLE_NUMBER>,
657    right_metadata: Metadata<K, PARENT_TABLE_NUMBER>,
658) -> (Y, Metadata<K, TABLE_NUMBER>) {
659    let left_metadata = u128::from(left_metadata);
660    let right_metadata = u128::from(right_metadata);
661
662    let parent_metadata_bits = metadata_size_bits(K, PARENT_TABLE_NUMBER);
663
664    // Part of the `right_bits` at the final offset of eventual `input_a`
665    let y_and_left_bits = y_size_bits(K) + parent_metadata_bits;
666    let right_bits_start_offset = u128::BITS as usize - parent_metadata_bits;
667
668    // Take only bytes where bits were set
669    let num_bytes_with_data =
670        (y_size_bits(K) + parent_metadata_bits * 2).div_ceil(u8::BITS as usize);
671
672    // Only supports `K` from 15 to 25 (otherwise math will not be correct when concatenating y,
673    // left metadata and right metadata)
674    let hash = {
675        // Collect `K` most significant bits of `y` at the final offset of eventual `input_a`
676        let y_bits = u128::from(y) << (u128::BITS as usize - y_size_bits(K));
677
678        // Move bits of `left_metadata` at the final offset of eventual `input_a`
679        let left_metadata_bits =
680            left_metadata << (u128::BITS as usize - parent_metadata_bits - y_size_bits(K));
681
682        // If `right_metadata` bits start to the left of the desired position in `input_a` move
683        // bits right, else move left
684        if right_bits_start_offset < y_and_left_bits {
685            let right_bits_pushed_into_input_b = y_and_left_bits - right_bits_start_offset;
686            // Collect bits of `right_metadata` that will fit into `input_a` at the final offset in
687            // eventual `input_a`
688            let right_bits_a = right_metadata >> right_bits_pushed_into_input_b;
689            let input_a = y_bits | left_metadata_bits | right_bits_a;
690            // Collect bits of `right_metadata` that will spill over into `input_b`
691            let input_b = right_metadata << (u128::BITS as usize - right_bits_pushed_into_input_b);
692
693            let input = [input_a.to_be_bytes(), input_b.to_be_bytes()];
694            let input_len =
695                size_of::<u128>() + right_bits_pushed_into_input_b.div_ceil(u8::BITS as usize);
696            ab_blake3::single_block_hash(&input.as_flattened()[..input_len])
697                .expect("Exactly a single block worth of bytes; qed")
698        } else {
699            let right_bits_a = right_metadata << (right_bits_start_offset - y_and_left_bits);
700            let input_a = y_bits | left_metadata_bits | right_bits_a;
701
702            ab_blake3::single_block_hash(&input_a.to_be_bytes()[..num_bytes_with_data])
703                .expect("Less than a single block worth of bytes; qed")
704        }
705    };
706
707    let y_output = Y::from(
708        u32::from_be_bytes([hash[0], hash[1], hash[2], hash[3]])
709            >> (u32::BITS as usize - y_size_bits(K)),
710    );
711
712    let metadata_size_bits = metadata_size_bits(K, TABLE_NUMBER);
713
714    let metadata = if TABLE_NUMBER < 4 {
715        (left_metadata << parent_metadata_bits) | right_metadata
716    } else if metadata_size_bits > 0 {
717        // For K up to 25 it is guaranteed that metadata + bit offset will always fit into u128.
718        // We collect the bytes necessary, potentially with extra bits at the start and end of the
719        // bytes that will be taken care of later.
720        let metadata = u128::from_be_bytes(
721            hash[y_size_bits(K) / u8::BITS as usize..][..size_of::<u128>()]
722                .try_into()
723                .expect("Always enough bits for any K; qed"),
724        );
725        // Remove extra bits at the beginning
726        let metadata = metadata << (y_size_bits(K) % u8::BITS as usize);
727        // Move bits into the correct location
728        metadata >> (u128::BITS as usize - metadata_size_bits)
729    } else {
730        0
731    };
732
733    (y_output, Metadata::from(metadata))
734}

736// TODO: This is actually using only pipelining rather than real SIMD (at least explicitly) due to:
737//  * https://github.com/rust-lang/portable-simd/issues/108
738//  * https://github.com/BLAKE3-team/BLAKE3/issues/478#issuecomment-3200106103
739#[cfg(any(feature = "alloc", test))]
740#[cfg_attr(feature = "no-panic", no_panic::no_panic)]
741fn compute_fn_simd<const K: u8, const TABLE_NUMBER: u8, const PARENT_TABLE_NUMBER: u8>(
742    left_ys: [Y; COMPUTE_FN_SIMD_FACTOR],
743    left_metadatas: [Metadata<K, PARENT_TABLE_NUMBER>; COMPUTE_FN_SIMD_FACTOR],
744    right_metadatas: [Metadata<K, PARENT_TABLE_NUMBER>; COMPUTE_FN_SIMD_FACTOR],
745) -> (
746    Simd<u32, COMPUTE_FN_SIMD_FACTOR>,
747    [Metadata<K, TABLE_NUMBER>; COMPUTE_FN_SIMD_FACTOR],
748) {
749    let parent_metadata_bits = metadata_size_bits(K, PARENT_TABLE_NUMBER);
750    let metadata_size_bits = metadata_size_bits(K, TABLE_NUMBER);
751
752    // TODO: `u128` is not supported as SIMD element yet, see
753    //  https://github.com/rust-lang/portable-simd/issues/108
754    let left_metadatas: [u128; COMPUTE_FN_SIMD_FACTOR] = seq!(N in 0..16 {
755        [
756        #(
757            u128::from(left_metadatas[N]),
758        )*
759        ]
760    });
761    let right_metadatas: [u128; COMPUTE_FN_SIMD_FACTOR] = seq!(N in 0..16 {
762        [
763        #(
764            u128::from(right_metadatas[N]),
765        )*
766        ]
767    });
768
769    // Part of the `right_bits` at the final offset of eventual `input_a`
770    let y_and_left_bits = y_size_bits(K) + parent_metadata_bits;
771    let right_bits_start_offset = u128::BITS as usize - parent_metadata_bits;
772
773    // Take only bytes where bits were set
774    let num_bytes_with_data =
775        (y_size_bits(K) + parent_metadata_bits * 2).div_ceil(u8::BITS as usize);
776
777    // Only supports `K` from 15 to 25 (otherwise math will not be correct when concatenating y,
778    // left metadata and right metadata)
779    // TODO: SIMD hashing once this is possible:
780    //  https://github.com/BLAKE3-team/BLAKE3/issues/478#issuecomment-3200106103
781    let hashes: [_; COMPUTE_FN_SIMD_FACTOR] = seq!(N in 0..16 {
782        [
783        #(
784        {
785            let y = left_ys[N];
786            let left_metadata = left_metadatas[N];
787            let right_metadata = right_metadatas[N];
788
789            // Collect `K` most significant bits of `y` at the final offset of eventual
790            // `input_a`
791            let y_bits = u128::from(y) << (u128::BITS as usize - y_size_bits(K));
792
793            // Move bits of `left_metadata` at the final offset of eventual `input_a`
794            let left_metadata_bits =
795                left_metadata << (u128::BITS as usize - parent_metadata_bits - y_size_bits(K));
796
797            // If `right_metadata` bits start to the left of the desired position in `input_a` move
798            // bits right, else move left
799            if right_bits_start_offset < y_and_left_bits {
800                let right_bits_pushed_into_input_b = y_and_left_bits - right_bits_start_offset;
801                // Collect bits of `right_metadata` that will fit into `input_a` at the final offset
802                // in eventual `input_a`
803                let right_bits_a = right_metadata >> right_bits_pushed_into_input_b;
804                let input_a = y_bits | left_metadata_bits | right_bits_a;
805                // Collect bits of `right_metadata` that will spill over into `input_b`
806                let input_b = right_metadata << (u128::BITS as usize - right_bits_pushed_into_input_b);
807
808                let input = [input_a.to_be_bytes(), input_b.to_be_bytes()];
809                let input_len =
810                    size_of::<u128>() + right_bits_pushed_into_input_b.div_ceil(u8::BITS as usize);
811                ab_blake3::single_block_hash(&input.as_flattened()[..input_len])
812                    .expect("Exactly a single block worth of bytes; qed")
813            } else {
814                let right_bits_a = right_metadata << (right_bits_start_offset - y_and_left_bits);
815                let input_a = y_bits | left_metadata_bits | right_bits_a;
816
817                ab_blake3::single_block_hash(&input_a.to_be_bytes()[..num_bytes_with_data])
818                    .expect("Exactly a single block worth of bytes; qed")
819            }
820        },
821        )*
822        ]
823    });
824
825    let y_outputs = Simd::from_array(
826        hashes.map(|hash| u32::from_be_bytes([hash[0], hash[1], hash[2], hash[3]])),
827    ) >> (u32::BITS - y_size_bits(K) as u32);
828
829    let metadatas = if TABLE_NUMBER < 4 {
830        seq!(N in 0..16 {
831            [
832            #(
833                Metadata::from((left_metadatas[N] << parent_metadata_bits) | right_metadatas[N]),
834            )*
835            ]
836        })
837    } else if metadata_size_bits > 0 {
838        // For K up to 25 it is guaranteed that metadata + bit offset will always fit into u128.
839        // We collect the bytes necessary, potentially with extra bits at the start and end of the
840        // bytes that will be taken care of later.
841        seq!(N in 0..16 {
842            [
843            #(
844            {
845                let metadata = u128::from_be_bytes(
846                    hashes[N][y_size_bits(K) / u8::BITS as usize..][..size_of::<u128>()]
847                        .try_into()
848                        .expect("Always enough bits for any K; qed"),
849                );
850                // Remove extra bits at the beginning
851                let metadata = metadata << (y_size_bits(K) % u8::BITS as usize);
852                // Move bits into the correct location
853                Metadata::from(metadata >> (u128::BITS as usize - metadata_size_bits))
854            },
855            )*
856            ]
857        })
858    } else {
859        [Metadata::default(); _]
860    };
861
862    (y_outputs, metadatas)
863}

952/// # Safety
953/// `matches` must contain positions that correspond to the parent table. `ys`, `position` and
954/// `metadatas` length must be at least the length of `matches`
955#[cfg(feature = "alloc")]
956#[inline(always)]
957#[cfg_attr(feature = "no-panic", no_panic::no_panic)]
958unsafe fn matches_to_results<const K: u8, const TABLE_NUMBER: u8, const PARENT_TABLE_NUMBER: u8>(
959    parent_table: &Table<K, PARENT_TABLE_NUMBER>,
960    matches: &[Match],
961    ys: &mut [MaybeUninit<Y>],
962    positions: &mut [MaybeUninit<[Position; 2]>],
963    metadatas: &mut [MaybeUninit<Metadata<K, TABLE_NUMBER>>],
964) where
965    Table<K, PARENT_TABLE_NUMBER>: NotLastTable,
966{
967    let (grouped_matches, other_matches) = matches.as_chunks::<COMPUTE_FN_SIMD_FACTOR>();
968    let grouped_matches_len = grouped_matches.as_flattened().len();
969    // SAFETY: Function contract guarantees that outputs are at least as long as `matches`
970    let (grouped_ys, other_ys) = unsafe { ys.split_at_mut_unchecked(grouped_matches_len) };
971    let grouped_ys = grouped_ys.as_chunks_mut::<COMPUTE_FN_SIMD_FACTOR>().0;
972    // SAFETY: Function contract guarantees that outputs are at least as long as `matches`
973    let (grouped_positions, other_positions) =
974        unsafe { positions.split_at_mut_unchecked(grouped_matches_len) };
975    let grouped_positions = grouped_positions
976        .as_chunks_mut::<COMPUTE_FN_SIMD_FACTOR>()
977        .0;
978    // SAFETY: Function contract guarantees that outputs are at least as long as `matches`
979    let (grouped_metadatas, other_metadatas) =
980        unsafe { metadatas.split_at_mut_unchecked(grouped_matches_len) };
981    let grouped_metadatas = grouped_metadatas
982        .as_chunks_mut::<COMPUTE_FN_SIMD_FACTOR>()
983        .0;
984
985    for (((grouped_matches, grouped_ys), grouped_positions), grouped_metadatas) in grouped_matches
986        .iter()
987        .zip(grouped_ys)
988        .zip(grouped_positions)
989        .zip(grouped_metadatas)
990    {
991        // SAFETY: Guaranteed by function contract
992        let (ys_group, positions_group, metadatas_group) =
993            unsafe { match_to_result_simd(parent_table, grouped_matches) };
994        let ys_group = Y::array_from_repr(ys_group.to_array());
995        grouped_ys.write_copy_of_slice(&ys_group);
996        grouped_positions.write_copy_of_slice(&positions_group);
997
998        // The last table doesn't have metadata
999        if metadata_size_bits(K, TABLE_NUMBER) > 0 {
1000            grouped_metadatas.write_copy_of_slice(&metadatas_group);
1001        }
1002    }
1003    for (((other_match, other_y), other_positions), other_metadata) in other_matches
1004        .iter()
1005        .zip(other_ys)
1006        .zip(other_positions)
1007        .zip(other_metadatas)
1008    {
1009        // SAFETY: Guaranteed by function contract
1010        let (y, p, metadata) = unsafe { match_to_result(parent_table, other_match) };
1011        other_y.write(y);
1012        other_positions.write(p);
1013        // The last table doesn't have metadata
1014        if metadata_size_bits(K, TABLE_NUMBER) > 0 {
1015            other_metadata.write(metadata);
1016        }
1017    }
1018}

1020/// Find matches between a pair of adjacent buckets of the parent table and turn them into the
1021/// results of the current table, returns the number of matches processed.
1022///
1023/// # Safety
1024/// Buckets must come from `parent_table`, `ys`, `positions` and `metadatas` must have at least as
1025/// many elements as there are matches in the pair of buckets (at most [`REDUCED_MATCHES_COUNT`]).
1026#[cfg(feature = "alloc")]
1027#[inline(always)]
1028#[cfg_attr(feature = "no-panic", no_panic::no_panic)]
1029unsafe fn bucket_pair_to_results<
1030    const K: u8,
1031    const TABLE_NUMBER: u8,
1032    const PARENT_TABLE_NUMBER: u8,
1033>(
1034    parent_table: &Table<K, PARENT_TABLE_NUMBER>,
1035    left_bucket_index: u32,
1036    [left_bucket, right_bucket]: &[[(Position, Y); REDUCED_BUCKET_SIZE]; 2],
1037    ys: &mut [MaybeUninit<Y>],
1038    positions: &mut [MaybeUninit<[Position; 2]>],
1039    metadatas: &mut [MaybeUninit<Metadata<K, TABLE_NUMBER>>],
1040) -> usize
1041where
1042    Table<K, PARENT_TABLE_NUMBER>: NotLastTable,
1043{
1044    let mut matches = [MaybeUninit::uninit(); _];

1070/// Subspace's little-endian s-bucket convention: maps a table-7 entry's `first_k_bits` to its
1071/// s-bucket, returning a value `>= Record::NUM_S_BUCKETS` for entries whose low `K - 16` bits are
1072/// set (those are discarded). Consensus verification derives the challenge from the s-bucket with
1073/// the same little-endian byte layout, so this convention is fixed.
1074#[cfg(feature = "alloc")]
1075#[inline(always)]
1076fn little_endian_s_bucket(first_k_bits: u32, k: u8) -> u32 {
1077    let low_bits = u32::from(k) - 16;
1078    if first_k_bits & ((1 << low_bits) - 1) != 0 {
1079        return u32::MAX;
1080    }
1081    let cs_lo = (first_k_bits >> (u32::from(k) - 8)) & 0xff;
1082    let cs_hi = (first_k_bits >> low_bits) & 0xff;
1083    cs_lo | (cs_hi << 8)
1084}

1086/// Find matches between a pair of adjacent buckets of the parent table, turn them into proof
1087/// targets of the last table and hand each of them over to `store_target`, which is called at most
1088/// [`REDUCED_MATCHES_COUNT`] times.
1089///
1090/// # Safety
1091/// Buckets must come from `parent_table`.
1092#[cfg(feature = "alloc")]
1093#[inline(always)]
1094#[cfg_attr(feature = "no-panic", no_panic::no_panic)]
1095unsafe fn bucket_pair_to_proof_targets<const K: u8>(
1096    parent_table: &Table<K, 6>,
1097    left_bucket_index: u32,
1098    [left_bucket, right_bucket]: &[[(Position, Y); REDUCED_BUCKET_SIZE]; 2],
1099    mut store_target: impl FnMut(u16, [Position; 2]),
1100) {
1101    let mut matches = [MaybeUninit::uninit(); _];
1102    // SAFETY: Positions are taken from `Table::buckets()` and correspond to initialized values
1103    let matches = unsafe {
1104        find_matches_in_buckets(left_bucket_index, left_bucket, right_bucket, &mut matches)
1105    };
1106    // Throw away some successful matches that are not that necessary
1107    let matches = &matches[..matches.len().min(REDUCED_MATCHES_COUNT)];
1108
1109    let (grouped_matches, other_matches) = matches.as_chunks::<COMPUTE_FN_SIMD_FACTOR>();
1110
1111    for grouped_matches in grouped_matches {
1112        // SAFETY: Matches come from the parent table
1113        let (ys_group, positions_group, _) =
1114            unsafe { match_to_result_simd::<_, 7, _>(parent_table, grouped_matches) };
1115
1116        let s_buckets = ys_group >> Simd::splat(u32::from(PARAM_EXT));
1117
1118        for (s_bucket, p) in s_buckets.to_array().into_iter().zip(positions_group) {
1119            let Ok(s_bucket) = u16::try_from(little_endian_s_bucket(s_bucket, K)) else {
1120                continue;
1121            };
1122
1123            store_target(s_bucket, p);
1124        }
1125    }
1126    for other_match in other_matches {
1127        // SAFETY: Matches come from the parent table
1128        let (y, p, _) = unsafe { match_to_result::<_, 7, _>(parent_table, other_match) };
1129
1130        let Ok(s_bucket) = u16::try_from(little_endian_s_bucket(y.first_k_bits(), K)) else {
1131            continue;
1132        };
1133
1134        store_target(s_bucket, p);
1135    }
1136}

1138/// Store a proof target unless a target for this s-bucket was already found
1139#[cfg(feature = "alloc")]
1140#[inline(always)]
1141#[cfg_attr(feature = "no-panic", no_panic::no_panic)]
1142fn store_proof_target(
1143    table_6_proof_targets: &mut [[Position; 2]; const { Record::NUM_S_BUCKETS }],
1144    s_bucket: u16,
1145    positions: [Position; 2],
1146) {
1147    // There is an s-bucket for every possible `u16`, which is what makes the indexing below
1148    // statically within bounds
1149    const {
1150        assert!(Record::NUM_S_BUCKETS == usize::from(u16::MAX) + 1);
1151    }
1152
1153    let target = &mut table_6_proof_targets[usize::from(s_bucket)];
1154    if target == &[Position::ZERO; 2] {
1155        *target = positions;
1156    }
1157}

1485                        unsafe { &mut *positions.get_unchecked(left_bucket_index).get() };

1487                    // at this time, and it is guaranteed to be in range
1488                    let metadatas =
1489                        unsafe { &mut *metadatas.get_unchecked(left_bucket_index).get() };
1490                    // SAFETY: This is the only place where `left_bucket_index`'s entry is accessed
1491                    // at this time, and it is guaranteed to be in range

1493                        &mut *global_results_counts.get_unchecked(left_bucket_index).get()

1519

1586            Box::<[[Position; 2]; const { Record::NUM_S_BUCKETS }]>::new_zeroed().assume_init()
1587        };
1588
1589        Self::create_proof_targets_internal(&parent_table, &mut table_6_proof_targets);
1590
1591        let parent_table = parent_table.prune();
1592
1593        (table_6_proof_targets, parent_table)
1594    }

1596    /// The part of [`Self::create_proof_targets()`] that works with already allocated memory
1597    #[cfg_attr(feature = "no-panic", no_panic::no_panic)]
1598    fn create_proof_targets_internal(
1599        parent_table: &Table<K, 6>,
1600        table_6_proof_targets: &mut [[Position; 2]; const { Record::NUM_S_BUCKETS }],
1601    ) {
1602        for (buckets, left_bucket_index) in parent_table.buckets().array_windows().zip(0..) {
1603            // SAFETY: Buckets are taken from `parent_table`
1604            unsafe {
1605                bucket_pair_to_proof_targets(
1606                    parent_table,
1607                    left_bucket_index,
1608                    buckets,
1609                    |s_bucket, positions| {
1610                        store_proof_target(table_6_proof_targets, s_bucket, positions);
1611                    },
1612                );
1613            }
1614        }
1615    }

1617    /// Almost the same as [`Self::create_proof_targets()`], but uses parallelism internally for
1618    /// better performance (though not efficiency of CPU and memory usage), if you create multiple
1619    /// tables in parallel, prefer this method for better overall performance.
1620    #[cfg(feature = "parallel")]
1621    pub(super) fn create_proof_targets_parallel(
1622        parent_table: Table<K, 6>,
1623    ) -> (
1624        Box<[[Position; 2]; const { Record::NUM_S_BUCKETS }]>,
1625        PrunedTable<K, 6>,
1626    )
1627    where
1628        Table<K, 6>: NotLastTable,
1629    {
1630        // SAFETY: Contents is `MaybeUninit`
1631        let buckets_positions = unsafe {
1632            Box::<[SyncUnsafeCell<[MaybeUninit<_>; REDUCED_MATCHES_COUNT]>; NUM_BUCKET_PAIRS::<K>]>::new_uninit().assume_init()
1633        };
1634        let global_results_counts =
1635            array::from_fn::<_, { NUM_BUCKET_PAIRS::<K> }, _>(|_| SyncUnsafeCell::new(0u16));
1636
1637        let buckets = parent_table.buckets();
1638        // Iterate over buckets in batches, such that a cache line worth of bytes is taken from
1639        // `global_results_counts` each time to avoid unnecessary false sharing
1640        let bucket_batch_size = CACHE_LINE_SIZE / size_of::<u16>();
1641        let bucket_batch_index = AtomicUsize::new(0);
1642
1643        rayon::broadcast(|_ctx| {
1644            loop {
1645                let bucket_batch_index = bucket_batch_index.fetch_add(1, Ordering::Relaxed);
1646
1647                let buckets_batch = buckets
1648                    .array_windows()
1649                    .enumerate()
1650                    .skip(bucket_batch_index * bucket_batch_size)
1651                    .take(bucket_batch_size);
1652
1653                if buckets_batch.is_empty() {
1654                    break;
1655                }
1656
1657                for (left_bucket_index, buckets) in buckets_batch {
1658                    // SAFETY: This is the only place where `left_bucket_index`'s entry is accessed
1659                    // at this time, and it is guaranteed to be in range
1660                    let buckets_positions =
1661                        unsafe { &mut *buckets_positions.get_unchecked(left_bucket_index).get() };
1662                    // SAFETY: This is the only place where `left_bucket_index`'s entry is accessed
1663                    // at this time, and it is guaranteed to be in range
1664                    let count = unsafe {
1665                        &mut *global_results_counts.get_unchecked(left_bucket_index).get()
1666                    };
1667
1668                    let mut num_targets = 0_usize;
1669                    let store_target = |s_bucket, positions| {
1670                        // SAFETY: Targets are stored at most `REDUCED_MATCHES_COUNT` times, which
1671                        // is the size of `buckets_positions`
1672                        unsafe {
1673                            hint::assert_unchecked(num_targets < REDUCED_MATCHES_COUNT);
1674                        }
1675                        buckets_positions[num_targets].write((s_bucket, positions));
1676                        num_targets += 1;
1677                    };
1678
1679                    // SAFETY: Buckets are taken from `parent_table`
1680                    unsafe {
1681                        bucket_pair_to_proof_targets(
1682                            &parent_table,
1683                            left_bucket_index as u32,
1684                            buckets,
1685                            store_target,
1686                        );
1687                    }
1688
1689                    *count = num_targets as u16;
1690                }
1691            }
1692        });
1693
1694        let parent_table = parent_table.prune();
1695
1696        let buckets_positions = strip_sync_unsafe_cell(buckets_positions);
1697        let global_results_counts = global_results_counts.map(SyncUnsafeCell::into_inner);
1698
1699        // SAFETY: Data structure filled with zeroes is a valid invariant
1700        let mut table_6_proof_targets = unsafe {
1701            Box::<[[Position; 2]; const { Record::NUM_S_BUCKETS }]>::new_zeroed().assume_init()
1702        };
1703
1704        // SAFETY: `global_results_counts` corresponds to the number of initialized targets
1705        unsafe {
1706            Self::merge_proof_targets(
1707                &buckets_positions,
1708                &global_results_counts,
1709                &mut table_6_proof_targets,
1710            );
1711        }
1712
1713        (table_6_proof_targets, parent_table)
1714    }

1716    /// Merge proof targets found by [`Self::create_proof_targets_parallel()`] in parallel.
1717    ///
1718    /// # Safety
1719    /// `counts` must be the number of initialized targets in each entry of `buckets_targets`.
1720    #[cfg(feature = "parallel")]
1721    #[cfg_attr(feature = "no-panic", no_panic::no_panic)]
1722    #[expect(clippy::type_complexity, reason = "Internal API")]
1723    unsafe fn merge_proof_targets(
1724        buckets_targets: &[[MaybeUninit<(u16, [Position; 2])>; REDUCED_MATCHES_COUNT];
1725             NUM_BUCKET_PAIRS::<K>],
1726        counts: &[u16; NUM_BUCKET_PAIRS::<K>],
1727        table_6_proof_targets: &mut [[Position; 2]; const { Record::NUM_S_BUCKETS }],
1728    ) {
1729        for (targets, &count) in buckets_targets.iter().zip(counts) {
1730            // SAFETY: Function contract guarantees that this many targets are initialized, hence
1731            // the count is also within bounds
1732            let targets = unsafe {
1733                targets
1734                    .get_unchecked(..usize::from(count))
1735                    .assume_init_ref()
1736            };
1737
1738            for &(s_bucket, positions) in targets {
1739                store_proof_target(table_6_proof_targets, s_bucket, positions);
1740            }
1741        }
1742    }
1743}
