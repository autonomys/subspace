4use crate::chiapos::constants::PARAM_BC;
5use crate::chiapos::table::REDUCED_BUCKET_SIZE;
6use crate::chiapos::table::types::{Position, R};
7
8pub(super) struct Rmap {
9    /// `0` is a sentinel value indicating no virtual pointer is stored yet.
10    ///
11    /// Physical pointer must be increased by `1` to get a virtual pointer before storing. Virtual
12    /// pointer must be decreased by `1` before reading to get a physical pointer.
13    virtual_pointers: [u16; PARAM_BC as usize],
14    positions: [[Position; 2]; REDUCED_BUCKET_SIZE],
15    next_physical_pointer: u16,
16}
17
18impl Rmap {
19    #[inline(always)]
20    pub(super) fn new() -> Self {
21        Self {
22            virtual_pointers: [0; _],
23            positions: [[Position::SENTINEL; 2]; _],
24            next_physical_pointer: 0,
25        }
26    }
27
28    /// # Safety
29    /// `r` must be in the range `0..PARAM_BC`, there must be at most [`REDUCED_BUCKET_SIZE`] items
30    /// inserted
31    #[inline(always)]
32    unsafe fn insertion_item(&mut self, r: R) -> &mut [Position; 2] {
33        // SAFETY: Guaranteed by function contract
34        let virtual_pointer = unsafe { self.virtual_pointers.get_unchecked_mut(usize::from(r)) };
35
36        if let Some(physical_pointer) = virtual_pointer.checked_sub(1) {
37            // SAFETY: Internal pointers are always valid
38            return unsafe { self.positions.get_unchecked_mut(physical_pointer as usize) };
39        }
40
41        let physical_pointer = self.next_physical_pointer;
42        self.next_physical_pointer += 1;
43        *virtual_pointer = physical_pointer + 1;
44
45        // SAFETY: It is guaranteed by the function contract that the number of added elements will
46        // never exceed `REDUCED_BUCKETS_SIZE`, hence allocated pointers will always be within
47        // bounds
48        unsafe { self.positions.get_unchecked_mut(physical_pointer as usize) }
49    }
50
51    /// # Safety
52    /// `r` must be in the range `0..PARAM_BC`, there must be at most [`REDUCED_BUCKET_SIZE`] items
53    /// inserted
54    #[inline(always)]
55    pub(super) unsafe fn add(&mut self, r: R, position: Position) {
56        // SAFETY: Guaranteed by function contract
57        let rmap_item = unsafe { self.insertion_item(r) };
58
59        // The same `r` can appear in the table multiple times, one duplicate is supported here
60        if rmap_item[0] == Position::SENTINEL {
61            rmap_item[0] = position;
62        } else if rmap_item[1] == Position::SENTINEL {
63            rmap_item[1] = position;
64        }
65    }
66
67    /// # Safety
68    /// `r` must be in the range `0..PARAM_BC`
69    #[inline(always)]
70    pub(super) unsafe fn get(&self, r: R) -> [Position; 2] {
71        // SAFETY: Guaranteed by function contract
72        let virtual_pointer = *unsafe { self.virtual_pointers.get_unchecked(usize::from(r)) };
73
74        if let Some(physical_pointer) = virtual_pointer.checked_sub(1) {
75            // SAFETY: Internal pointers are always valid
76            *unsafe { self.positions.get_unchecked(physical_pointer as usize) }
77        } else {
78            [Position::SENTINEL; 2]
79        }
80    }
81}
