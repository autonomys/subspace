1use crate::chiapos::constants::{PARAM_B, PARAM_C, PARAM_M};
2use crate::chiapos::table::types::R;
3use core::array;
4use core::ops::{Add, Sub};
5use core::simd::prelude::*;
6use core::simd::{Mask, Simd, SimdElement};

8/// Number of left targets calculated at a time
9#[cfg(feature = "alloc")]
10const SIMD_FACTOR: usize = 16;
11
12/// Calculates left targets of the left bucket with a given parity.
13///
14/// Two `y`s match when `r` of the right entry is one of the [`PARAM_M`] targets calculated here
15/// from `r` of the left entry.
16#[derive(Debug)]
17pub(super) struct LeftTargets {
18    /// [`Self::SQUARES`] of the parity this instance was created for
19    squares: &'static [u16; const { usize::from(PARAM_M) }],
20}
21
22impl LeftTargets {
23    /// `(2 * m + parity)^2 % PARAM_C` for each `m` of each parity.
24    ///
25    /// This is the only part of a target that doesn't depend on `r`, the rest is a handful of
26    /// cheap SIMD instructions, which is why targets are calculated on the fly instead of being
27    /// read from a large precomputed table (the GPU implementation calculates them too).
28    const SQUARES: [[u16; const { usize::from(PARAM_M) }]; 2] =
29        [Self::squares(0), Self::squares(1)];
30
31    /// A single row of [`Self::SQUARES`].
32    ///
33    /// `array::from_fn()` would have been nicer, but it needs a `const` closure to be usable here.
34    const fn squares(parity: u16) -> [u16; const { usize::from(PARAM_M) }] {
35        let mut squares = [0; _];
36
37        let mut m = 0;
38        while m < PARAM_M {
39            let value = 2 * m + parity;
40            squares[usize::from(m)] = (value * value) % PARAM_C;
41            m += 1;
42        }
43
44        squares
45    }
46
47    #[inline(always)]
48    #[cfg_attr(feature = "no-panic", no_panic::no_panic)]
49    pub(super) fn new(odd_parity: bool) -> Self {
50        Self {
51            squares: &Self::SQUARES[usize::from(odd_parity)],
52        }
53    }
54
55    /// `a + b` modulo `modulus`.
56    ///
57    /// Both `a` and `b` must be below `modulus`, which puts the sum below `modulus * 2` and turns
58    /// the modulo into a single conditional subtraction.
59    ///
60    /// Writing `%` instead is not an option. LLVM derives the necessary bound for scalars only:
61    /// the range of a splat is whatever the scalar's range was, but `LazyValueInfo` has no rule
62    /// for `shufflevector`, and `InstCombine` canonicalizes a masked splat into exactly that
63    /// shape, so no way of writing this expression reaches the conditional subtraction.
64    /// `std::hint::assert_unchecked()` doesn't help either. What is emitted instead is the generic
65    /// reciprocal multiply, which for `u16x16` is 7 instructions with two multiplications against
66    /// 4 here on x86-64, and 16 with six against 7 on aarch64.
67    // TODO: Use `%` once LLVM looks through splats when computing ranges:
68    //  https://github.com/llvm/llvm-project/issues/223468
69    #[inline(always)]
70    #[cfg_attr(feature = "no-panic", no_panic::no_panic)]
71    fn add_mod<T, const N: usize>(a: Simd<T, N>, b: Simd<T, N>, modulus: Simd<T, N>) -> Simd<T, N>
72    where
73        T: SimdElement + Default,
74        Simd<T, N>: Add<Output = Simd<T, N>>
75            + Sub<Output = Simd<T, N>>
76            + SimdPartialOrd<Mask = Mask<T::Mask, N>>,
77    {
78        let sum = a + b;
79
80        sum - modulus.simd_le(sum).select(modulus, Simd::default())
81    }
82
83    /// Calculate all [`PARAM_M`] targets of `r` at once
84    #[cfg(feature = "alloc")]
85    #[inline(always)]
86    #[cfg_attr(feature = "no-panic", no_panic::no_panic)]
87    pub(super) fn calculate(&self, r: R) -> [R; const { usize::from(PARAM_M) }] {
88        let r = u16::from(r);
89        let c = Simd::splat(r / PARAM_C);
90        let d = Simd::splat(r % PARAM_C);
91        let param_b = Simd::splat(PARAM_B);
92        let param_c = Simd::splat(PARAM_C);
93
94        let mut targets = [0u16; _];
95
96        for (index, (targets, squares)) in targets
97            .as_chunks_mut::<SIMD_FACTOR>()
98            .0
99            .iter_mut()
100            .zip(self.squares.as_chunks::<SIMD_FACTOR>().0)
101            .enumerate()
102        {
103            let ms = Simd::splat((index * SIMD_FACTOR) as u16)
104                + Simd::from_array(array::from_fn(|offset| offset as u16));
105
106            let target_c = Self::add_mod(c, ms, param_b);
107            let target_d = Self::add_mod(d, Simd::from_array(*squares), param_c);
108
109            *targets = (target_c * param_c + target_d).to_array();
110        }
111
112        R::array_from_repr(targets)
113    }
114
115    /// Calculate all [`PARAM_M`] targets of `r` at once and check whether `r_target` is one of
116    /// them.
117    ///
118    /// This is what [`Self::calculate()`] is for verification, where targets are only compared to
119    /// a single `r` and never stored.
120    #[inline(always)]
121    #[cfg_attr(feature = "no-panic", no_panic::no_panic)]
122    pub(super) fn contains(&self, r: R, r_target: R) -> bool {
123        let r = u16::from(r);
124        let c = Simd::splat(r / PARAM_C);
125        let d = Simd::splat(r % PARAM_C);
126        let param_b = Simd::splat(PARAM_B);
127        let param_c = Simd::splat(PARAM_C);
128
129        let ms = Simd::from_array(array::from_fn(|m| m as u16));
130
131        let target_c = Self::add_mod(c, ms, param_b);
132        let target_d = Self::add_mod(d, Simd::from_array(*self.squares), param_c);
133
134        (target_c * param_c + target_d)
135            .simd_eq(Simd::splat(u16::from(r_target)))
136            .any()
137    }
138}
